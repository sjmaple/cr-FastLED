156 // (#2558) mLegacySettings.mWhiteCfg is now a fl::variant<Empty, Rgbw, Rgbww>;
157 // assigning Rgbw selects the 4-channel alternative. The legacy

410 // Not virtual on AVR (#4788). Its only overrides are the RGBW/WS2816
411 // wrappers, which forward to a disabled inner controller; on AVR the base
412 // version saves and restores the wrapper's own dither instead. Custom AVR
413 // controllers cannot hook these, or size().
414 VIRTUAL_IF_NOT_AVR void* beginShowLeds(int size) FL_NO_EXCEPT {
415 FASTLED_UNUSED(size);
416 // By default, emit an integer. This integer will, by default, be passed back.
417 // If you override beginShowLeds() then
418 // you should also override endShowLeds() to match the return state.
419 //
420 // For async led controllers this should be used as a sync point to block
421 // the caller until the leds from the last draw frame have completed drawing.
422 // for each controller:
423 // beginShowLeds();
424 // for each controller:
425 // showLeds();
426 // for each controller:
427 // endShowLeds();
428 uintptr_t d = getDither();
429 void* out = fl::int_to_ptr<void>(d);
430 return out;
431 }
432
433 VIRTUAL_IF_NOT_AVR void endShowLeds(void* data) FL_NO_EXCEPT {
434 // By default recieves the integer that beginShowLeds() emitted.
435 //For async controllers this should be used to signal the controller
436 // to begin transmitting the current frame to the leds.
437 uintptr_t d = fl::ptr_to_int(data);
438 setDither(static_cast<fl::u8>(d));
439 }
