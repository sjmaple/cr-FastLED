17// unconditionally re-enabling interrupts, so we don't clobber the
18// caller's IRQ-disabled state if we're invoked from inside another
19// critical section. Inline asm avoids needing the CMSIS header.
20static inline uint32_t fl_save_primask_disable() {
21 uint32_t pm;

24}

48void ObjectFLEDDmaManager::waitForCompletion() {
49 // Spin-wait for DMA completion with periodic delays.
50 while (!dma3.complete()) {
51 delayMicroseconds(10);
52 }
53 // #3416 OF-MED-6: dma3.complete() reflects DMA major-loop done but
54 // does NOT include the WS2812 reset latch period. If the caller is
55 // about to destroy the LED buffer or hand the line off to another
56 // peripheral, returning here can corrupt the final frame's tail
57 // bits. The destructor wait path adds its own latch margin (see
58 // ObjectFLED::~ObjectFLED) but defensive callers should also gate
59 // on busy()==0 which factors in LATCH_DELAY. Documented here so
60 // the next reader doesn't mistake "DMA done" for "frame done".
61}

63bool ObjectFLEDDmaManager::ensureBitdata(uint32_t numbytes) {
64 const uint32_t words =
65 objectfled::grownBitdataWords(bitdataWords, numbytes, BYTES_PER_DMA);
66 if (words == bitdataWords && bitdata != nullptr) {
67 return true;
68 }
69 // Over-allocate and align the start to a 32-byte cache line. The size
70 // is a multiple of 128 B, so arm_dcache_flush_delete() never touches a
71 // line shared with other heap data.
72 // Free the old (too small) buffer first: it can't serve this frame, and
73 // freeing it lets the allocator reuse that space for the larger block.
74 fl::free(mBitdataAlloc);
75 mBitdataAlloc = nullptr;
76 bitdata = nullptr;
77 bitdataWords = 0;
78 void* block = fl::malloc(words * sizeof(uint32_t) + 31);
79 if (block == nullptr) {
80 return false; // caller drops the frame
81 }
82 mBitdataAlloc = block;
83 bitdata = reinterpret_cast<uint32_t*>(
84 (reinterpret_cast<uintptr_t>(block) + 31) & ~uintptr_t(31));
85 bitdataWords = words;
86 return true;
87}
