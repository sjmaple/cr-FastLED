190// SFINAE: detect if T is default-constructible
191template <typename T, typename = void>
192struct has_default_ctor : fl::false_type {};
193
194template <typename T>
195struct has_default_ctor<T, decltype(void(T()))> : fl::true_type {};
196
197// SFINAE: detect if T is copy-constructible
198template <typename T, typename = void>
199struct has_copy_ctor : fl::false_type {};
200
201template <typename T>
202struct has_copy_ctor<T, decltype(void(T(fl::declval<const T&>())))> : fl::true_type {};
203
204// Default construct: available when T has default ctor
205template <typename T>
206typename fl::enable_if<has_default_ctor<T>::value, void(*)(void*)>::type
207get_default_construct_fn() FL_NO_EXCEPT {
208 return [](void* ptr) FL_NO_EXCEPT { new (ptr) T(); };
209}
210
211// Default construct: nullptr when T has no default ctor
212template <typename T>
213typename fl::enable_if<!has_default_ctor<T>::value, void(*)(void*)>::type
214get_default_construct_fn() FL_NO_EXCEPT {
215 return nullptr;
216}
217
218// Copy construct: available when T has copy ctor
219template <typename T>
220typename fl::enable_if<has_copy_ctor<T>::value, void(*)(void*, const void*)>::type
221get_copy_construct_fn() FL_NO_EXCEPT {
222 return [](void* dst, const void* src) FL_NO_EXCEPT {
223 new (dst) T(*static_cast<const T*>(src));
224 };
225}

234// SFINAE: detect if T is move-constructible
235template <typename T, typename = void>
236struct has_move_ctor : fl::false_type {};

238template <typename T>
239struct has_move_ctor<T, decltype(void(T(fl::declval<T&&>())))> : fl::true_type {};

241// SFINAE: detect if T is swappable (needs move-constructible + move-assignable)
242template <typename T, typename = void>
243struct is_swappable : fl::false_type {};
244
245template <typename T>
246struct is_swappable<T, decltype(void(
247 fl::declval<T&>() = fl::declval<T&&>()
248))> : has_move_ctor<T> {};
249
250// Move construct: available
251template <typename T>
252typename fl::enable_if<has_move_ctor<T>::value, void(*)(void*, void*)>::type
253get_move_construct_fn() FL_NO_EXCEPT {
254 return [](void* dst, void* src) FL_NO_EXCEPT {
255 new (dst) T(static_cast<T&&>(*static_cast<T*>(src)));
256 };
257}
258
259template <typename T>
260typename fl::enable_if<!has_move_ctor<T>::value, void(*)(void*, void*)>::type
261get_move_construct_fn() FL_NO_EXCEPT {
262 return nullptr;
263}
264
265// Swap: available when swappable
266template <typename T>
267typename fl::enable_if<is_swappable<T>::value, void(*)(void*, void*)>::type
268get_swap_fn() FL_NO_EXCEPT {
269 return [](void* a, void* b) FL_NO_EXCEPT {
270 T& ta = *static_cast<T*>(a);
271 T& tb = *static_cast<T*>(b);
272 T tmp(static_cast<T&&>(ta));
273 ta = static_cast<T&&>(tb);
274 tb = static_cast<T&&>(tmp);
275 };
276}

278template <typename T>
279typename fl::enable_if<!is_swappable<T>::value, void(*)(void*, void*)>::type
280get_swap_fn() FL_NO_EXCEPT {
281 return nullptr;

290template <typename T>
291const vector_element_ops* vector_element_ops_for() FL_NO_EXCEPT {
292 // Trivially copyable types use the fast path (mOps == nullptr)
293 if (fl::is_trivially_copyable<T>::value) {
294 return nullptr;
295 }
296
297 static const vector_element_ops ops = {
298 detail::get_copy_construct_fn<T>(), // copy_construct
299 detail::get_move_construct_fn<T>(), // move_construct
300 detail::get_default_construct_fn<T>(), // default_construct
301 detail::get_swap_fn<T>(), // swap_elements
302 // destroy_n
303 [](void* first, fl::size count) FL_NO_EXCEPT {
304 T* p = static_cast<T*>(first);
305 for (fl::size i = 0; i < count; ++i) {
306 p[i].~T();
307 }
308 }
309 };
310 return &ops;
311}
