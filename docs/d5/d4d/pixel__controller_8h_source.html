244 // controller's `d`/`e` state and brightness. `NO_DITHERING=1` selects

250 void init_binary_dithering() FL_NO_EXCEPT {
251 fl::u8 frame = 0;
252 if (fl::Dither::kEnabled) {
253#if FL_PLATFORM_HAS_TINY_MEMORY
254 // Tiny targets retain their one-byte local phase counter.
255 static fl::u8 R = 0; // okay static in header: preserves the tiny-target footprint
256 frame = ++R;
257#else
258 // Other targets share a phase once per logical frame.
259 frame = fl::detail::ditherFrame();
260#endif
261 }
262 fl::Dither::init(d, e, mColorAdjustment.premixed.raw, frame);
263 }

564 FASTLED_FORCE_INLINE void loadAndScaleRGBW(
565 const Rgbw& rgbw, fl::u8 *b0_out, fl::u8 *b1_out,
566 fl::u8 *b2_out, fl::u8 *b3_out) FL_NO_EXCEPT {
567#ifdef FL_IS_AVR
568 // Don't do RGBW conversion for AVR, just set the W pixel to black.
569 fl::u8 out[4] = {
570 // Get the pixels in native order.
571 loadAndScale0(),
572 loadAndScale1(),
573 loadAndScale2(),
574 0,
575 };
576 EOrderW w_placement = rgbw.w_placement;
577 // Apply w-component insertion.
578 fl::rgbw_partial_reorder(
579 w_placement, out[0], out[1], out[2],
580 0, // Pre-ordered RGB data with a 0 white component.
581 b0_out, b1_out, b2_out, b3_out);
582#else
583 const fl::u8 b0_index = RGB_BYTE0(RGB_ORDER); // Needed to re-order RGB back into led native order.
584 const fl::u8 b1_index = RGB_BYTE1(RGB_ORDER);
585 const fl::u8 b2_index = RGB_BYTE2(RGB_ORDER);
586 // Source-order RGB after white extraction, then the wire reorder.
587 CRGB rgb;
588 fl::u8 w = 0;
589 loadAndScaleRGBWUnordered(rgbw, &rgb.r, &rgb.g, &rgb.b, &w);
590 // Now finish the ordering so that the output is in the native led order for all of RGBW.
591 fl::rgbw_partial_reorder(
592 rgbw.w_placement,
593 rgb.raw[b0_index], // in-place re-ordering for the RGB data.
594 rgb.raw[b1_index],
595 rgb.raw[b2_index],
596 w, // The white component is not ordered in this call.
597 b0_out, b1_out, b2_out, b3_out); // RGBW data now in total native led order.
598#endif
599 }
