45void Pacifica::draw(DrawContext ctx) FL_NO_EXCEPT {
46 fl::span<CRGB> leds = ctx.leds;
47 fl::u32 now = ctx.now;
48 if (leds.empty() || mNumLeds == 0) {
49 return;
50 }
51
52 // Update the hue each time through the loop
53 fl::u32 ms = now;
54 fl::u32 deltams = ms - sLastms;
55 sLastms = ms;
56 u16 speedfactor1 = beatsin16(3, 179, 269);
57 u16 speedfactor2 = beatsin16(4, 179, 269);
58 fl::u32 deltams1 = (deltams * speedfactor1) / 256;
59 fl::u32 deltams2 = (deltams * speedfactor2) / 256;
60 fl::u32 deltams21 = (deltams1 + deltams2) / 2;
61 sCIStart1 += (deltams1 * beatsin88(1011, 10, 13));
62 sCIStart2 -= (deltams21 * beatsin88(777, 8, 11));
63 sCIStart3 -= (deltams1 * beatsin88(501, 5, 7));
64 sCIStart4 -= (deltams2 * beatsin88(257, 4, 6));
65
66 // Clear out the LED array to a dim background blue-green
67 fill_solid(leds, CRGB(2, 6, 10));
68
69 // Render each of four layers, with different scales and speeds, that vary
70 // over time
71 pacifica_one_layer(leds, pacifica_palette_1, sCIStart1,
72 beatsin16(3, 11 * 256, 14 * 256), beatsin8(10, 70, 130),
73 0 - beat16(301));
74 pacifica_one_layer(leds, pacifica_palette_2, sCIStart2,
75 beatsin16(4, 6 * 256, 9 * 256), beatsin8(17, 40, 80),
76 beat16(401));
77 pacifica_one_layer(leds, pacifica_palette_3, sCIStart3, 6 * 256,
78 beatsin8(9, 10, 38), 0 - beat16(503));
79 pacifica_one_layer(leds, pacifica_palette_3, sCIStart4, 5 * 256,
80 beatsin8(8, 10, 28), beat16(601));
81
82 // Add brighter 'whitecaps' where the waves lines up more
83 pacifica_add_whitecaps(leds);
84
85 // Deepen the blues and greens a bit
86 pacifica_deepen_colors(leds);
87}

89// Add one layer of waves into the led array
90void Pacifica::pacifica_one_layer(fl::span<CRGB> leds, CRGBPalette16 &p,
91 u16 cistart, u16 wavescale,
92 u8 bri, u16 ioff) FL_NO_EXCEPT {
93 u16 ci = cistart;
94 u16 waveangle = ioff;
95 u16 wavescale_half = (wavescale / 2) + 20;
96 for (u16 i = 0; i < mNumLeds; i++) {
97 waveangle += 250;
98 u16 s16 = sin16(waveangle) + 32768;
99 u16 cs = scale16(s16, wavescale_half) + wavescale_half;
100 ci += cs;
101 u16 sindex16 = sin16(ci) + 32768;
102 u8 sindex8 = scale16(sindex16, 240);
103 CRGB c = ColorFromPalette(p, sindex8, bri, LINEARBLEND);
104 leds[i] += c;
105 }
106}

108// Add extra 'white' to areas where the four layers of light have lined up
109// brightly
110void Pacifica::pacifica_add_whitecaps(fl::span<CRGB> leds) FL_NO_EXCEPT {
111 u8 basethreshold = beatsin8(9, 55, 65);
112 u8 wave = beat8(7);
113
114 for (u16 i = 0; i < mNumLeds; i++) {
115 u8 threshold = scale8(sin8(wave), 20) + basethreshold;
116 wave += 7;
117 u8 l = leds[i].getAverageLight();
118 if (l > threshold) {
119 u8 overage = l - threshold;
120 u8 overage2 = qadd8(overage, overage);
121 leds[i] += CRGB(overage, overage2, qadd8(overage2, overage2));
122 }
123 }
124}
