76void run(fl::u32 microseconds, ExecFlags flags) FL_NO_EXCEPT {
77 // Re-entrancy guard: detect if run is called from within run
78 bool& running = SingletonThreadLocal<bool>::instance();
79 if (running) {
80 FL_WARN_ONCE("task::run re-entrancy detected, skipping nested call");
81 return;
82 }
83 running = true;
84 auto guard = fl::make_scope_exit([&running]() { running = false; });
85
86 const bool do_tasks = flags & ExecFlags::TASKS;
87 const bool do_coroutines = flags & ExecFlags::COROUTINES;
88 const bool do_system = flags & ExecFlags::SYSTEM;
89
90 // Calculate start time with rollover protection
91 fl::u32 begin_time = fl::micros();
92
93 // Lambda to get elapsed time (rollover-safe)
94 auto elapsed = [begin_time]() {
95 return fl::micros() - begin_time;
96 };
97
98 // Lambda to get remaining time until deadline expires
99 auto remaining = [elapsed, microseconds]() -> fl::u32 {
100 fl::u32 e = elapsed();
101 if (e >= microseconds) {
102 return 0;
103 }
104 return microseconds - e;
105 };
106
107 // Lambda to check if deadline has expired
108 auto expired = [remaining]() {
109 return remaining() == 0;
110 };
111
112 do {
113 // TASKS: Scheduler (fl::task timers) + Executor (fetch, HTTP server, audio)
114 if (do_tasks) {
115 Scheduler::instance().update();
116 Executor::instance().update_all();
117 }
118
119 // SYSTEM: OS-level yield.
120 //
121 // When the caller provided a non-zero microseconds budget we treat
122 // this as an explicit spin-wait on hardware (typical DMA / driver
123 // wait loops). In that case we MUST use a deep yield (>= 1 FreeRTOS
124 // tick on ESP32) so that lower-priority network tasks — WiFi,
125 // Ethernet lwIP, etc. — actually get CPU time. Previously this
126 // path was gated on an active WiFi mode check, which missed
127 // Ethernet-only deployments and also the pre-connection window
128 // where `esp_wifi_get_mode` still returns WIFI_MODE_NULL. That
129 // regression manifested as the ESP32-S3 I2S-vs-websockets
130 // starvation reported in https://github.com/FastLED/FastLED/issues/2254
131 // (Issue 1).
132 //
133 // When microseconds == 0 the caller wants the cheapest possible
134 // yield (we are between other pumped subsystems and will loop
135 // again immediately), so we keep the lightweight fl::yield() path.
136 //
137 // The sleep is clamped to whatever is left of the caller's budget,
138 // mirroring the COROUTINES branch below. This used to be a hardcoded
139 // pumpCoroutines(1000) that ignored `microseconds` entirely, so a
140 // caller asking for run(250, SYSTEM) slept 1000 us and overshot its
141 // own deadline by 750 us on every platform where pumpCoroutines
142 // honors the requested duration (host/stub -- CoroutineRunner::run).
143 //
144 // ESP32 behavior is unchanged: pumpCoroutines() there converts us to
145 // FreeRTOS ticks and floors at one tick for ANY non-zero request
146 // (coroutine_esp32.impl.hpp, the `if (ticks == 0) ticks = 1` branch
147 // added for #2254), so 250 and 1000 both yield exactly one tick. The
148 // deep-yield guarantee #2254 depends on still holds.
149 if (do_system) {
150 if (microseconds > 0) {
151 const fl::u32 time_left = remaining();
152 if (time_left) {
153 fl::platforms::ICoroutineRuntime::instance().pumpCoroutines(
154 fl::min(1000u, time_left));
155 }
156 } else {
157 fl::yield();
158 }
159 }
160
161 // COROUTINES: Platform cooperative coroutines (pumpCoroutines)
162 if (do_coroutines) {
163 auto time_left = remaining();
164 if (time_left) {
165 fl::u32 sleep_us = fl::min(1000u, time_left);
166 fl::platforms::ICoroutineRuntime::instance().pumpCoroutines(sleep_us);
167 }
168 }
169 } while (!expired());
170}

172void yield_system(fl::u32 microseconds) FL_NO_EXCEPT {
173 // Mirrors the SYSTEM branch of run() without naming Scheduler/Executor.
174 const fl::u32 begin_time = fl::micros();
175 do {
176 if (microseconds > 0) {
177 const fl::u32 e = fl::micros() - begin_time;
178 if (e < microseconds) {
179 fl::platforms::ICoroutineRuntime::instance().pumpCoroutines(
180 fl::min(1000u, microseconds - e));
181 }
182 } else {
183 fl::yield();
184 }
185 } while (fl::micros() - begin_time < microseconds);
186}
