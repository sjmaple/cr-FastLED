142 constexpr PowerModelRGB toRGB() const FL_NO_EXCEPT {
143 // Distribute white-channel mW evenly across R/G/B. Clamp to 255 (u8
144 // max) on overflow — a wrapping static_cast<u8> here would make the
145 // brightness limiter *less* conservative on the highest-draw configs
146 // (where it most needs to be conservative). The worst-case
147 // under-budget per channel from the floor division is 2 mW.
148 return PowerModelRGB(
149 static_cast<fl::u8>(
150 (red_mW + (white_mW + warm_white_mW) / 3) > 255
151 ? 255 : (red_mW + (white_mW + warm_white_mW) / 3)),
152 static_cast<fl::u8>(
153 (green_mW + (white_mW + warm_white_mW) / 3) > 255
154 ? 255 : (green_mW + (white_mW + warm_white_mW) / 3)),
155 static_cast<fl::u8>(
156 (blue_mW + (white_mW + warm_white_mW) / 3) > 255
157 ? 255 : (blue_mW + (white_mW + warm_white_mW) / 3)),
158 dark_mW,
159 exponent);
160 }
161};
