60 u32 high_min, u32 high_max,

76inline int flexPwmBitFromHigh(u32 high_ns, const ChipsetTiming4Phase& timing) FL_NO_EXCEPT {
77 if (high_ns >= timing.t0h_min_ns && high_ns <= timing.t0h_max_ns) {
78 return 0;
79 }
80 if (high_ns >= timing.t1h_min_ns && high_ns <= timing.t1h_max_ns) {
81 return 1;
82 }
83 return -1;
84}
