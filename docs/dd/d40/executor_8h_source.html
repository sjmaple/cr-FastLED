192template<typename T>
193PromiseResult<T> await_top_level(Promise<T> p) FL_NO_EXCEPT {
194 // Handle invalid promises
195 if (!p.valid()) {
196 return PromiseResult<T>(Error("Invalid promise"));
197 }
198
199 // If already completed, return immediately
200 if (p.is_completed()) {
201 if (p.is_resolved()) {
202 return PromiseResult<T>(p.value());
203 } else {
204 return PromiseResult<T>(p.error());
205 }
206 }
207
208 // Track recursion depth to prevent infinite loops
209 int& await_depth = detail::await_depth_tls();
210 if (await_depth > 10) {
211 return PromiseResult<T>(Error("await_top_level recursion limit exceeded - possible infinite loop"));
212 }
213
214 ++await_depth;
215
216 // Wait for promise to complete while pumping async tasks
217 int pump_count = 0;
218 const int max_pump_iterations = 10000; // Safety limit
219
220 while (!p.is_completed() && pump_count < max_pump_iterations) {
221 // Update the promise first (in case it's not managed by async system)
222 p.update();
223
224 // Check if completed after update
225 if (p.is_completed()) {
226 break;
227 }
228
229 // Platform-agnostic async pump and yield
230 run(1000);
231
232 ++pump_count;
233 }
234
235 --await_depth;
236
237 // Check for timeout
238 if (pump_count >= max_pump_iterations) {
239 return PromiseResult<T>(Error("await_top_level timeout - promise did not complete"));
240 }
241
242 // Return the result
243 if (p.is_resolved()) {
244 return PromiseResult<T>(p.value());
245 } else {
246 return PromiseResult<T>(p.error());
247 }
248}

253// ============================================================================
254// Await API Comparison: await() vs await_top_level()
255// ============================================================================
256//
257// FastLED provides two await functions for different contexts:
258//
259// 1. fl::task::await() - For coroutines (ESP32/Host only)
260// - Zero CPU overhead (true OS-level blocking)
261// - Must be called from fl::task::coroutine() context
262// - Suspends coroutine thread until promise completes
263// - Platform support: ESP32 (FreeRTOS), Host/Stub (std::thread)
264//
265// 2. fl::task::await_top_level() - For main loop/top-level code (All platforms)
266// - High CPU usage (busy-wait with run(1000))
267// - Safe to call from Arduino loop() or main()
268// - NOT safe in promise callbacks or nested async operations
269// - Platform support: All platforms
270//
271// **When to use which?**
272// - In fl::task::coroutine(): Use fl::task::await() for zero CPU waste
273// - In Arduino loop() or main(): Use fl::task::await_top_level()
274// - Other platforms without coroutine support: Use fl::task::await_top_level()
275//
276// ============================================================================
