70// fl::strncpy - bounded string copy that ALWAYS null-terminates.
71// Copies at most n chars from src, then writes a terminator at
72// dest[copied] (copied <= n). Pass n = capacity - 1 (dest sized >= n+1).
73// Differs from ::strncpy, which leaves dest unterminated when
74// strlen(src) >= n. Remaining destination bytes up to n are zero-padded.
75template <typename FlLibcTag = void>
76FASTLED_FORCE_INLINE char* strncpy(char* dest, const char* src, size_t n) FL_NO_EXCEPT { return libc_impl::strncpy(dest, src, n); }
