15fl::result<u32, DecodeError>
16decodeFlexPwmEdges(const ChipsetTiming4Phase& timing,
17 fl::span<const EdgeTime> edges,
18 fl::span<u8> bytes_out) FL_NO_EXCEPT {
19 if (edges.size() == 0 || bytes_out.size() == 0) {
20 return fl::result<u32, DecodeError>::success(0);
21 }
22
23 // Pre-pass: the adaptive midpoint from the observed HIGH widths.
24 u32 high_min = 0xFFFFFFFFu;
25 u32 high_max = 0u;
26 u32 high_samples = 0;
27 for (size_t k = 0; k + 1 < edges.size(); k += 2) {
28 if (!edges[k].high) continue;
29 const u32 h = edges[k].ns;
30 if (!flexPwmIsMidpointSample(h)) continue;
31 if (h < high_min) high_min = h;
32 if (h > high_max) high_max = h;
33 ++high_samples;
34 }
35 const u32 midpoint =
36 flexPwmAdaptiveMidpointNs(timing, high_min, high_max, high_samples);
37
38 u32 byte_index = 0;
39 u8 current_byte = 0;
40 u32 bit_count = 0;
41 u32 error_count = 0;
42 u32 total_bits = 0;
43
44 size_t i = 0;
45 while (i < edges.size()) {
46 if (!edges[i].high) {
47 ++i; // polarity error: resync on the next HIGH
48 continue;
49 }
50 const u32 high_ns = edges[i].ns;
51 int bit = -1;
52 if (i + 1 == edges.size()) {
53 // The frame's last HIGH: its LOW is the reset, so classify by
54 // HIGH width alone, like a mid-frame bit. (Stopping at the last
55 // edge dropped the frame's final bit: LSB 1 decoded as 0.)
56 bit = flexPwmBitFromMidpoint(high_ns, midpoint);
57 i += 1;
58 } else if (!edges[i + 1].high) {
59 bit = flexPwmBitFromMidpoint(high_ns, midpoint);
60 i += 2;
61 } else {
62 bit = flexPwmBitFromHigh(high_ns, timing);
63 i += 1;
64 }
65 ++total_bits;
66 if (bit < 0) {
67 ++error_count;
68 continue;
69 }
70 current_byte = static_cast<u8>((current_byte << 1) | static_cast<u8>(bit));
71 if (++bit_count == 8) {
72 if (byte_index >= bytes_out.size()) {
73 return fl::result<u32, DecodeError>::failure(
74 DecodeError::BUFFER_OVERFLOW);
75 }
76 bytes_out[byte_index++] = current_byte;
77 current_byte = 0;
78 bit_count = 0;
79 }
80 }
81
82 // Partial final byte, left-aligned.
83 if (bit_count > 0 && byte_index < bytes_out.size()) {
84 bytes_out[byte_index++] = static_cast<u8>(current_byte << (8 - bit_count));
85 }
86 if (total_bits > 0 && (error_count * 10) > total_bits) {
87 return fl::result<u32, DecodeError>::failure(DecodeError::HIGH_ERROR_RATE);
88 }
89 return fl::result<u32, DecodeError>::success(byte_index);
90}

150FASTLED_FORCE_INLINE void Ws2812StreamDecoder::classify(u32 high_ns, Mode mode) FL_NO_EXCEPT {
151 ++mStats.bits;
152 if (mode == Mode::FromHigh) {
153 const int b = flexPwmBitFromHigh(high_ns, mTiming);
154 record(high_ns, mode, b >= 0);
155 if (b < 0) {
156 ++mStats.errors;
157 return;
158 }
159 appendBit(b);
160 return;
161 }
162 const u32 dist = (high_ns > mMidpoint) ? high_ns - mMidpoint : mMidpoint - high_ns;
163 if (dist <= kExceptionBandNs) {
164 record(high_ns, mode, true);
165 }
166 appendBit(flexPwmBitFromMidpoint(high_ns, mMidpoint));
167}

198FASTLED_FORCE_INLINE void Ws2812StreamDecoder::emitPair(u16 rise, u16 fall, bool has_low,
199 u32 low_ns) FL_NO_EXCEPT {
200 const u32 high_ns = flexPwmTickDeltaNs(rise, fall, mNsPerTickQ16);
201 // The batch pre-pass samples HIGHs at even edge indices that are not
202 // the last edge (the final pair, emitted by flush(), never is).
203 if ((mEdgeIndex & 1u) == 0 && flexPwmIsMidpointSample(high_ns)) {
204 if (high_ns < mHighMin) mHighMin = high_ns;
205 if (high_ns > mHighMax) mHighMax = high_ns;
206 ++mHighSamples;
207 }
208 emitEdge(true, high_ns);
209 if (has_low) {
210 emitEdge(false, low_ns);
211 pushBit(high_ns, Mode::Midpoint);
212 } else {
213 pushBit(high_ns, Mode::FromHigh);
214 }
215}

217// Hot path: runs in the capture DMA ISR, once per ring half.
218FL_OPTIMIZE_FUNCTION void Ws2812StreamDecoder::push(fl::span<const u16> captures) FL_NO_EXCEPT {
219 const size_t pair_count = captures.size() / 2u;
220 for (size_t p = 0; p < pair_count; ++p) {
221 const u16 rise = captures[2 * p];
222 const u16 fall = captures[2 * p + 1];
223 ++mStats.pairs;
224 if (mHavePrev) {
225 const u32 low_ns = flexPwmTickDeltaNs(mPrevFall, rise, mNsPerTickQ16);
226 const bool has_low = low_ns <= kFlexPwmGapNs;
227 // Leading phantom pairs (#3219, #3409): the arm sequence can
228 // latch a stray pair followed by an idle gap. Drop any leading
229 // pair whose LOW is a gap.
230 if (mStarted || has_low) {
231 mStarted = true;
232 emitPair(mPrevRise, mPrevFall, has_low, low_ns);
233 }
234 }
235 mPrevRise = rise;
236 mPrevFall = fall;
237 mHavePrev = true;
238 }

241void Ws2812StreamDecoder::flush() FL_NO_EXCEPT {
242 if (mFlushed) {
243 return;
244 }
245 mFlushed = true;
246 if (mHavePrev) {
247 // The final pair's HIGH is the last edge: not a pre-pass sample, and
248 // classified by the midpoint like the batch decoder's last edge.
249 const u32 high_ns = flexPwmTickDeltaNs(mPrevRise, mPrevFall, mNsPerTickQ16);
250 emitEdge(true, high_ns);
251 pushBit(high_ns, Mode::Midpoint);
252 mHavePrev = false;
253 }
254 if (!mCalibrated) {
255 calibrate();
256 }

259void Ws2812StreamDecoder::finish(const ChipsetTiming4Phase& timing) FL_NO_EXCEPT {
260 flush();
261 mPartialWritten = false;
262 mStats.gapBitShift = false;
263 if (mBitCount > 0 && mStats.fullBytes < mOut.size()) {
264 mOut[mStats.fullBytes] = static_cast<u8>(mCurByte << (8 - mBitCount));
265 mPartialWritten = true;
266 }
267
268 const u32 frame_mid =
269 flexPwmAdaptiveMidpointNs(timing, mHighMin, mHighMax, mHighSamples);
270 mStats.frameMidpointNs = frame_mid;
271 const u32 drift = (frame_mid > mMidpoint) ? frame_mid - mMidpoint : mMidpoint - frame_mid;
272 bool inexact = drift > kExceptionBandNs || mStats.exceptionsDropped > 0;
273 for (u32 k = 0; k < mExceptionCount; ++k) {
274 const Exception& e = mExceptions[k];
275 int b;
276 if (e.mode == Mode::FromHigh) {
277 b = flexPwmBitFromHigh(e.high_ns, timing);
278 if ((b >= 0) != e.stream_valid) {
279 inexact = true; // the bit would be dropped / kept instead
280 mStats.gapBitShift = true;
281 continue;
282 }
283 if (b < 0) {
284 continue;
285 }
286 } else {
287 b = flexPwmBitFromMidpoint(e.high_ns, frame_mid);
288 }
289 const u32 byte = e.bit_pos / 8u;
290 if (byte >= mOut.size()) {
291 continue;
292 }
293 const u8 mask = static_cast<u8>(0x80u >> (e.bit_pos % 8u));
294 mOut[byte] = b ? static_cast<u8>(mOut[byte] | mask)
295 : static_cast<u8>(mOut[byte] & ~mask);
296 }
297 mStats.inexact = inexact;
298}

300fl::result<u32, DecodeError> Ws2812StreamDecoder::copyTo(fl::span<u8> out) const FL_NO_EXCEPT {
301 if (mStats.overrun) {
302 return fl::result<u32, DecodeError>::failure(DecodeError::CAPTURE_OVERRUN);
303 }
304 if (mStats.gapBitShift) {
305 // The capture-time timing kept or dropped a gap bit that the decode
306 // timing would not: every later bit is shifted. Pass the decode
307 // timing as RxConfig::stream_timing.
308 return fl::result<u32, DecodeError>::failure(DecodeError::INVALID_ARGUMENT);
309 }
310 if (mStats.pairs == 0 || out.size() == 0) {
311 return fl::result<u32, DecodeError>::success(0);
312 }
313 const u32 full = mStats.fullBytes;
314 if (full > out.size() || full > mOut.size()) {
315 return fl::result<u32, DecodeError>::failure(DecodeError::BUFFER_OVERFLOW);
316 }
317 u32 n = full;
318 if (mPartialWritten && n < out.size()) {
319 ++n;
320 }
321 for (u32 k = 0; k < n; ++k) {
322 out[k] = mOut[k];
323 }
324 if (mStats.bits > 0 && (mStats.errors * 10) > mStats.bits) {
325 return fl::result<u32, DecodeError>::failure(DecodeError::HIGH_ERROR_RATE);
326 }
327 return fl::result<u32, DecodeError>::success(n);
328}
