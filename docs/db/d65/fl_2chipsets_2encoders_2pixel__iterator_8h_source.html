247// PixelIterator is turns a PixelController<> into a concrete object that can be used to iterate
248// over pixels and transform them into driver data. See PixelController<>::as_iterator() for how
249// to create a PixelIterator.
250// Note: This is designed for micro-controllers with a lot of memory. DO NOT use this in the core library
251// as a PixelIterator consumes a *lot* more instruction data than an instance of PixelController<RGB_ORDER>.
252// This iterator is designed for code in src/platforms/**.
253class PixelIterator {
254 public:
255 template<typename PixelControllerT>
256 PixelIterator(PixelControllerT* pc, const Rgbw& rgbw,
257 Rgbww rgbww = RgbwwInvalid::value()) FL_NO_EXCEPT
258 : mPixelController(pc), mRgbw(rgbw), mRgbww(rgbww) {
259 // Manually build up a vtable.
260 // Wait... what? Stupid nerds trying to show off how smart they are...
261 // Why not just use a virtual function?!
262 //
263 // Before you think this then you should know that the alternative straight
264 // forward way is to have a virtual interface class that PixelController inherits from.
265 // ...and that was already tried. And if you try to do this yourself
266 // this then let me tell you what is going to happen...
267 //
268 // EVERY SINGLE PLATFORM THAT HAS A COMPILED BINARY SIZE CHECK WILL IMMEDIATELY
269 // FAIL AS THE BINARY BLOWS UP BY 10-30%!!! It doesn't matter if only one PixelController
270 // with a vtable is used, gcc seems not to de-virtualize the calls. And we really care
271 // about binary size since FastLED needs to run on those tiny little microcontrollers like
272 // the Attiny85 (and family) which are in the sub $1 range used for commercial products.
273 //
274 // So to satisfy these tight memory requirements we make the dynamic dispatch used in PixelIterator
275 // an optional zero-cost abstraction which doesn't affect the binary size for platforms that
276 // don't use it. So that's why we are using this manual construction of the vtable that is built
277 // up using template magic. If your platform has lots of memory then you'll gladly trade
278 // a sliver of memory for the convenience of having a concrete implementation of
279 // PixelController that you can use without having to make all your driver code a template.
280 //
281 // Btw, this pattern in C++ is called the "type-erasure pattern". It allows non virtual
282 // polymorphism by leveraging the C++ template system to ensure type safety.
283 typedef PixelControllerVtable<PixelControllerT> Vtable;
284 mLoadAndScaleRGBW = RgbwLoadBinder<PixelControllerT>::ordered();
285 mLoadAndScaleRGBWW = RgbwLoadBinder<PixelControllerT>::orderedWW();
286 mLoadAndScaleRGBWUnordered = RgbwLoadBinder<PixelControllerT>::unordered();
287 mLoadAndScaleRGBWWUnordered = RgbwLoadBinder<PixelControllerT>::unorderedWW();
288 mLoadRawPremixed = RawPremixBinder<PixelControllerT>::get();
289 mSourceOrder = RawPremixBinder<PixelControllerT>::order();
290 if (mLoadRawPremixed != nullptr) {
291 mRgbwConvert = mRgbw.active() ? detail::rgbw_conversion() : nullptr;
292 mRgbwwConvert = mRgbww.active() ? detail::rgbww_conversion() : nullptr;
293 refreshWireOrder();
294 }
295 mLoadAndScaleRGB = &Vtable::loadAndScaleRGB;
296#if !FL_PLATFORM_HAS_TINY_MEMORY
297 mLoadAndScaleRGB16 = WideLoadBinder<PixelControllerT>::get();
298#endif
299 // NOTE: mLoadAndScale_APA102_HD removed - use fl::loadAndScale_APA102_HD<RGB_ORDER>() from apa102.h encoder
300 // NOTE: mLoadAndScale_WS2816_HD removed - use fl::loadAndScale_WS2816_HD<RGB_ORDER>() from ws2816.h encoder
301 mStepDithering = &Vtable::stepDithering;
302 mAdvanceData = &Vtable::advanceData;
303 mSize = &Vtable::size;
304 mHas = &Vtable::has;
305 #if FASTLED_HD_COLOR_MIXING
306 mLoadRGBScaleAndBrightness = &Vtable::loadRGBScaleAndBrightness;
307 mGetHdScale = &Vtable::getHdScale;
308 #endif
309 }

337 void loadAndScaleRGBW(u8 *b0_out, u8 *b1_out, u8 *b2_out, u8 *w_out) FL_NO_EXCEPT {
338 if (mLoadRawPremixed != nullptr) {
339 // Type-erased RGBW: the conversion is installed by active Rgbw
340 // construction / setRgbw(), so RGB-only programs never link it
341 // (#4795). Cached at construction (the Rgbw already exists then).
342 u8 raw[3];
343 u8 premix[3];
344 mLoadRawPremixed(mPixelController, raw, premix);
345 if (mRgbwConvert != nullptr) {
346 mRgbwConvert(mRgbw, raw, premix, mWireOrder, b0_out, b1_out, b2_out, w_out);
347 return;
348 }
349 // Not installed: only an inactive Rgbw can reach here. Exactly what
350 // rgb_2_rgbw(kRGBWInvalid) produced: premix-scaled RGB, white off,
351 // then the W placement.
352 u8 c[3];
353 u8 w = 0;
354 rgb_2_rgbw_null_white_pixel(mRgbw.white_color_temp, raw[0], raw[1], raw[2],
355 premix[0], premix[1], premix[2],
356 &c[0], &c[1], &c[2], &w);
357 rgbw_partial_reorder(mRgbw.w_placement, c[mWireOrder[0]], c[mWireOrder[1]],
358 c[mWireOrder[2]], w, b0_out, b1_out, b2_out, w_out);
359 return;
360 }
361#if defined(FL_IS_AVR)
362 // AVR keeps the order-free thunk (no type-erased conversion there).
363 if (mLoadAndScaleRGBWUnordered != nullptr) {
364 // Source-order RGB plus W, then the same two steps the templated
365 // `loadAndScaleRGBW` takes: permute RGB, place W.
366 u8 c[3];
367 u8 w = 0;
368 mLoadAndScaleRGBWUnordered(mPixelController, mRgbw, &c[0], &c[1], &c[2], &w);
369 rgbw_partial_reorder(mRgbw.w_placement, c[mOrder[0]], c[mOrder[1]], c[mOrder[2]], w,
370 b0_out, b1_out, b2_out, w_out);
371 return;
372 }
373#endif
374 mLoadAndScaleRGBW(mPixelController, mRgbw, b0_out, b1_out, b2_out, w_out);

739 template <typename CONTAINER_UIN8_T>
740 void writeHD108(CONTAINER_UIN8_T* out, bool managed = false) FL_NO_EXCEPT {
741 auto back_ins = fl::back_inserter(*out);
742
743 // One loop for both sources. This used to be two: the managed path
744 // written out by hand here, and the 8-bit path routed through
745 // `encodeHD108` over `makeScaledPixelRangeRGB`. Each carried its own
746 // copy of the framing -- the 8-byte start frame, the per-LED emit of
747 // header + three big-endian channels, the latch -- and together they
748 // made this the largest encoder body in a Blink build (481 B on
749 // esp32dev, ahead of `writeWS2812`, the only one Blink runs). The only
750 // thing that differed was where the 16-bit channels came from, so
751 // that is now the only branch (FastLED#4402).
752 //
753 // Byte-identical to the iterator path by construction:
754 // `ScaledPixelIteratorRGB::advance()` is exactly `has(1)`,
755 // `loadAndScaleRGB` into wire order, `stepDithering`, `advanceData`,
756 // once per pixel; the bytes it hands `encodeHD108` are captured before
757 // the step, which is what this loop does too.
758 //
759 // Inline rather than an `encodeHD108_wide` in hd108.h: that header is
760 // included *by* this one, before `PixelIterator` exists, and a
761 // concrete `PixelIterator&` parameter is not a dependent type -- so
762 // its body would be checked there and fail (#4326).
763#if FL_PLATFORM_HAS_TINY_MEMORY
764 // No wide source exists on TINY, so the flag selects nothing there.
765 FL_UNUSED(managed);
766#endif
767 for (int i = 0; i < 8; i++) {
768 *back_ins++ = 0x00; // start frame
769 }
770 // `hd108BrightnessHeader` discards its argument and pins every gain
771 // at 31, so this is 0xFF 0xFF whatever is passed (#4402).
772 u8 f0, f1;
773 hd108BrightnessHeader(0, &f0, &f1);
774 fl::size num_leds = 0;
775 while (has(1)) {
776 u16 r16, g16, b16;
777#if !FL_PLATFORM_HAS_TINY_MEMORY
778 if (managed) {
779 // The managed source has already quantized its device drive
780 // once, to 16 bits. `hd108GammaCorrect` exists to widen an
781 // 8-bit pixel to the 16 this chipset carries; a fixed 2.8
782 // curve on top of a solved drive is the second shaping stage
783 // B1 and section 6 of the spec forbid after the device
784 // solve, and unlike UCS7604's `mGamma.value_or(2.8f)` this
785 // one is hardcoded -- no caller could have chosen otherwise.
786 loadAndScaleRGB16(&r16, &g16, &b16);
787 } else
788#endif
789 {
790 u8 b0, b1, b2;
791 loadAndScaleRGB(&b0, &b1, &b2); // wire order
792 r16 = hd108GammaCorrect(b0);
793 g16 = hd108GammaCorrect(b1);
794 b16 = hd108GammaCorrect(b2);
795 }
796 *back_ins++ = f0;
797 *back_ins++ = f1;
798 *back_ins++ = static_cast<u8>(r16 >> 8);
799 *back_ins++ = static_cast<u8>(r16 & 0xFF);
800 *back_ins++ = static_cast<u8>(g16 >> 8);
801 *back_ins++ = static_cast<u8>(g16 & 0xFF);
802 *back_ins++ = static_cast<u8>(b16 >> 8);
803 *back_ins++ = static_cast<u8>(b16 & 0xFF);
804 stepDithering();
805 advanceData();
806 ++num_leds;
807 }
808 const fl::size latch = num_leds / 2 + 4;
809 for (fl::size i = 0; i < latch; i++) {
810 *back_ins++ = 0xFF; // end frame
811 }
812 }

934inline void ScaledPixelIteratorRGB16::advance() FL_NO_EXCEPT {
935 if (!mPixels) {
936 mHasValue = false;
937 return;
938 }
939
940 if (mPixels->has(1)) {
941 // Wire-ordered, colour-corrected, brightness-scaled pixel bytes.
942 //
943 // This used to branch on FASTLED_HD_COLOR_MIXING and call
944 // loadRGBScaleAndBrightness() here, treating its three outputs as the
945 // pixel. They are not the pixel: that call returns
946 // ColorAdjustment::color -- the correction *scale* at full
947 // brightness, a per-strip constant -- for a caller that will apply it
948 // to a pixel itself. So every LED came out the same colour and the
949 // frame never reached the wire (#4323). WS2816, the only user of this
950 // range, showed one flat tint on every non-AVR build.
951 //
952 // loadAndScaleRGB() is the call that loads the pixel. It applies
953 // `premixed`, which already carries brightness, so there is no
954 // separate brightness step left to do. The old branch existed to
955 // defer brightness to 16-bit precision, but it was producing a
956 // constant rather than extra precision, so nothing is given up by
957 // dropping it. Doing that properly needs a primitive that loads the
958 // pixel scaled at full brightness, which PixelIterator does not
959 // expose today.

964 // colour-managed source instead quantizes its s16.16 drive straight
965 // to 16 bits, which is what lets a 16-bit encoder reach more than 256
966 // levels.

984 } else {
