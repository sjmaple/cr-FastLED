13// Compile-time regression for FastLED#4827: legacy sketches call libc memory
14// functions unqualified on CRGB*. These must stay unambiguous on every board.
15static void legacyLibcOnCrgb() {
16 CRGB src[4] = {CRGB(1, 2, 3), CRGB(4, 5, 6), CRGB(7, 8, 9), CRGB(10, 11, 12)};
17 CRGB* dst = static_cast<CRGB*>(malloc(sizeof(CRGB) * 4)); // ok bare allocation
18 if (!dst) {
19 return;
20 }
21 memset(dst, 0, sizeof(CRGB) * 4);
22 memcpy(dst, src, sizeof(CRGB) * 4);
23 memmove(dst + 1, dst, sizeof(CRGB) * 3);
24 if (memcmp(dst, src, sizeof(CRGB)) != 0) {
25 Serial.println("legacy libc copy mismatch");
26 }
27 free(dst); // ok bare allocation
28}
