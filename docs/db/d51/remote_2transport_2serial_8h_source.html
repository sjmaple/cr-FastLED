119inline fl::function<fl::optional<fl::json>()>
120createSerialRequestSource(const char* prefix = "") FL_NO_EXCEPT {
121 return [prefix]() -> fl::optional<fl::json> {
122 // Non-blocking check - any data available?
123 int avail = fl::available();
124 if (avail <= 0) {
125 return fl::nullopt;
126 }
127
128 // Data available - read a complete line.
129 // On Arduino platforms, fl::readLine() delegates to Serial.readStringUntil()
130 // which uses yield() (immediate context switch) for fast USB CDC multi-packet
131 // assembly. On other platforms, falls back to character-by-character reading.
132 auto line = fl::readLine('\n', '\r', fl::optional<u32>(1000));
133 if (!line.has_value() || line->empty()) {
134 return fl::nullopt;
135 }
136
137 // Use string_view for zero-copy prefix stripping and trimming
138 fl::string_view view = *line;
139
140 // Strip prefix if present
141 if (prefix && prefix[0] != '\0') {
142 if (view.starts_with(prefix)) {
143 view.remove_prefix(fl::strlen(prefix));
144 }
145 }
146
147 // Trim leading whitespace
148 while (!view.empty() && fl::isspace(view.front())) {
149 view.remove_prefix(1);
150 }
151
152 // Trim trailing whitespace
153 while (!view.empty() && fl::isspace(view.back())) {
154 view.remove_suffix(1);
155 }
156
157 // Only parse if input starts with '{'
158 if (view.empty() || view[0] != '{') {
159 // A short line here is ordinary serial noise -- boot banners,
160 // stray newlines -- and warning about it would drown the log.
161 // A long one is almost certainly a request that arrived damaged,
162 // and dropping that silently is expensive to debug: the caller
163 // sees no reply and no error, indistinguishable from an idle
164 // link, while the device keeps answering everything else. That
165 // is exactly how a receive-buffer truncation presented in
166 // FastLED#3956 -- 735 bytes transmitted, nothing returned, and
167 // no layer saying anything.
168 // Measure what arrived, not what survived normalisation: the
169 // view above has had its prefix stripped and whitespace trimmed,
170 // so a 64-byte line can fall under the threshold and stay silent,
171 // and the reported count would understate what was received.
172 if (line->size() >= kDroppedRequestWarnBytes) {
173 FL_WARN("[RPC] discarded a " << (static_cast<unsigned>(line->size())) << " byte line that does not begin with '{'; a request that arrived truncated looks exactly like this");
174 }
175 return fl::nullopt;
176 }
177
178 // Single copy when parsing JSON (unavoidable - JSON needs owned string)
179 fl::string input(view);
180 return fl::json::parse(input);
181 };
182}

201inline fl::ResponseStreamSink
202createSerialResponseStreamSink(const char* prefix = "REMOTE: ") FL_NO_EXCEPT {
203 return [prefix](fl::JsonStreamCallback writeJson) {
204 SerialWriter serial;
205 if (prefix && prefix[0] != '\0') {
206 serial.write(prefix, fl::strlen(prefix));
207 }
208
209 fl::JsonStreamWriter writer([&serial](const char* data, fl::size len) {
210 serial.write(data, len);
211 });
212 if (writeJson) {
213 writeJson(writer);
214 }
215 writer.flush();
216 serial.write("\n", 1);
217 fl::flush();
218 };
219}

246template<typename SerialIn>
247fl::optional<fl::string> readSerialStringUntil(SerialIn& serial, char delimiter, char skipChar, fl::optional<u32> timeoutMs) FL_NO_EXCEPT {
248 // Follows Arduino Serial.readStringUntil() API - blocks until delimiter found
249 fl::sstream buffer;
250
251 u32 startTime = fl::millis();
252
253 // Read characters until we find delimiter or timeout
254 while (true) {
255 // Check timeout (only if timeout is set)
256 if (timeoutMs.has_value()) {
257 if (fl::millis() - startTime >= timeoutMs.value()) {
258 // Timeout occurred - return nullopt
259 return fl::nullopt;
260 }
261 }
262
263 // Try to read next character
264 int c = serial.read();
265
266 // Handle -1 (no data available) like Arduino's timedRead():
267 // Keep trying until timeout (or forever if no timeout set)
268 if (c == -1) {
269 // Brief 1us yield instead of 1ms delay for fast USB CDC polling.
270 fl::delayMicroseconds(1);
271 continue;
272 }
273
274 // Found delimiter - complete
275 if (c == delimiter) {
276 break;
277 }
278
279 // Skip specified character (e.g., '\r' for cross-platform line endings)
280 if (c == skipChar) {
281 continue;
282 }
283
284 // Valid character - add to buffer
285 buffer << static_cast<char>(c);
286 }
287
288 // Convert to string and trim whitespace
289 fl::string result = buffer.str();
290 result.trim();
291 return result;
292}
