57void Blend2d::draw(DrawContext context) FL_NO_EXCEPT {
58 mFrame->clear();
59 mFrameTransform->clear();
60
61 // Draw each layer in reverse order and applying the blending.
62 bool first = true;
63 for (auto it = mLayers.begin(); it != mLayers.end(); ++it) {
64 DrawContext tmp_ctx = context;
65 tmp_ctx.leds = mFrame->rgb();
66 auto &fx = it->fx;
67 fx->draw(tmp_ctx);
68 DrawMode mode = first ? DrawMode::DRAW_MODE_OVERWRITE
69 : DrawMode::DRAW_MODE_BLEND_BY_MAX_BRIGHTNESS;
70 first = false;
71 // Apply the blur effect per effect.
72 u8 blur_amount = it->blur_amount;
73 if (blur_amount > 0) {
74 const XYMap &xyMap = fx->getXYMap();
75 u8 blur_passes = fl::max(1, it->blur_passes);
76 for (u8 i = 0; i < blur_passes; ++i) {
77 // Apply the blur effect
78 blur2d(mFrame->rgb().data(), mXyMap.getWidth(), mXyMap.getHeight(),
79 blur_amount, xyMap);
80 }
81 }
82 mFrame->draw(mFrameTransform->rgb(), mode);
83 }
84
85 if (mGlobalBlurAmount > 0) {
86 // Apply the blur effect
87 u16 width = mXyMap.getWidth();
88 u16 height = mXyMap.getHeight();
89 XYMap rect = XYMap::constructRectangularGrid(width, height);
90 fl::span<CRGB> rgb = mFrameTransform->rgb();
91 u8 blur_passes = fl::max(1, mGlobalBlurPasses);
92 for (u8 i = 0; i < blur_passes; ++i) {
93 // Apply the blur effect
94 blur2d(rgb, width, height, mGlobalBlurAmount, rect);
95 }
96 }
97
98 // Copy the final result to the output
99 // memcpy(mFrameTransform->rgb(), context.leds, sizeof(CRGB) *
100 // mXyMap.getTotal());
101 mFrameTransform->drawXY(context.leds, mXyMap,
102 DrawMode::DRAW_MODE_OVERWRITE);
103}
