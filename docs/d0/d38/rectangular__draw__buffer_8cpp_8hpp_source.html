51bool RectangularDrawBuffer::onQueuingDone() FL_NO_EXCEPT {
52 if (mQueueState == QUEUE_DONE) {
53 return false;
54 }
55 mQueueState = QUEUE_DONE;
56 mDrawListChangedThisFrame = mDrawList != mPrevDrawList;
57 // iterator through the current draw objects and calculate the total
58 // number of bytes (representing RGB or RGBW) that will be drawn this frame.
59 u32 total_bytes = 0;
60 u32 max_bytes_in_strip = 0;
61 u32 num_strips = 0;
62 getBlockInfo(&num_strips, &max_bytes_in_strip, &total_bytes);
63 if (total_bytes > mAllLedsBufferUint8Size) {
64 u8 *old_ptr = mAllLedsBufferUint8.release();
65 fl::PSRamAllocator<u8>::Free(old_ptr);
66 u8 *ptr = fl::PSRamAllocator<u8>::Alloc(total_bytes);
67 mAllLedsBufferUint8.reset(ptr);
68 }
69 mAllLedsBufferUint8Size = total_bytes;
70 u32 offset = 0;
71 for (auto it = mDrawList.begin(); it != mDrawList.end(); ++it) {
72 u8 pin = it->mPin;
73 span<u8> slice(mAllLedsBufferUint8.get() + offset,
74 max_bytes_in_strip);
75 mPinToLedSegment[pin] = slice;
76 offset += max_bytes_in_strip;
77 }
78 return true;
79}
