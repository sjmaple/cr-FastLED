408fl::json WLED::getState() const FL_NO_EXCEPT {

410 state.set("on", mWledOn);

418 // Nightlight object

434 state.set("playlist", getPlaylistConfig());
435 }
436
437 // Segments
438 if (!mSegments.empty()) {
439 fl::json segments = fl::json::array();
440 for (const auto& seg : mSegments) {
441 fl::json segJson = fl::json::object();
