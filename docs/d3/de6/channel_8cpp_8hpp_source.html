35class ReorderingPixelIteratorAny {
36 private:
39 static fl::vector<CRGB>& getReorderBufferTLS() FL_NO_EXCEPT {
40 return SingletonThreadLocal<fl::vector<CRGB>>::instance();
41 }
42 fl::optional<PixelController<RGB, 1, 0xFFFFFFFF>> mAddressedController;
43 PixelIteratorAny mPixelIterator;
44#if FL_COLOR_PROFILE_RUNTIME
45 // Present only when a colour profile is bound. The managed source keeps
46 // a reference to the controller it wraps, so it has to outlive the
47 // iterator built over it -- both live here, for the frame.
48 //
49 // Raw storage rather than the types themselves, because naming them here
50 // is what made the whole colour pipeline reachable from `show()` and cost
51 // every sketch ~3.5 KB of flash. `colorPipelineHooks()` builds into this,
52 // and only `setColorProfile` installs those hooks -- so a program that
53 // never binds a profile never references the pipeline and the linker
54 // drops it.
55 FL_ALIGNAS(8) unsigned char mManagedSourceStorage[kColorPipelineSourceStorage];
56 FL_ALIGNAS(8) unsigned char mManagedIteratorStorage[kColorPipelineIteratorStorage];
57 PixelIterator* mManagedIterator = nullptr;
58#endif
59
60 public:
67 ReorderingPixelIteratorAny(
68 PixelController<RGB, 1, 0xFFFFFFFF>& pixels,
69 const XYMap* addressing,
70 EOrder rgbOrder,
71 Rgbw rgbw,
72 Rgbww rgbww,
73 const fl::string& channelName,
74#if FL_COLOR_PIPELINE_SHARED
75 const StreamingPipelineQ16* pipeline,
76#else
77 StreamingPipelineQ16* pipeline,
78#endif
79 u8 ditherPhase, u8 brightness) FL_NO_EXCEPT
80 : mPixelIterator(pixels, rgbOrder, rgbw, rgbww) {
81 FL_UNUSED(pipeline);
82 FL_UNUSED(ditherPhase); // read only by the colour-managed source
83 FL_UNUSED(brightness); // read only by the colour-managed source
84 FL_UNUSED(channelName); // only consumed by FL_ERROR, a no-op on small platforms
85
86 // Apply addressing transformation if configured
87 if (addressing) {
88 u16 numLeds = pixels.size();
89 u16 width = addressing->getWidth();
90 u16 height = addressing->getHeight();
91 u16 expectedLeds = width * height;
92
93 // Validate that XYMap dimensions match channel LED count
94 if (expectedLeds != numLeds) {
95 FL_ERROR("Channel '" << channelName << "': XYMap dimensions (" << width << "x" << height << "=" << expectedLeds << ") don't match LED count (" << numLeds << "). Addressing transformation may produce unexpected results.");
96 }
97
98 // Cast mData to CRGB array
99 const CRGB* pixelArray = (const CRGB*)pixels.mData;
100
101 // Get thread-local buffer (reuses allocation if size unchanged)
102 fl::vector<CRGB>& buffer = getReorderBufferTLS();
103 buffer.clear();
104 buffer.resize(numLeds);
105
106 // Fill buffer by mapping each physical index to its source
107 for (u16 physicalIdx = 0; physicalIdx < numLeds; physicalIdx++) {
108 u16 x = physicalIdx % width;
109 u16 y = physicalIdx / width;
110 u16 sourceIdx = addressing->mapToIndex(x, y);
111 buffer[physicalIdx] = (sourceIdx < numLeds) ? pixelArray[sourceIdx] : CRGB::Black;
112 }
113
114 // Construct PixelController with reordered buffer
115 mAddressedController.emplace(
116 buffer.data(), buffer.size(),
117 pixels.mColorAdjustment, DISABLE_DITHER);
118 mPixelIterator = PixelIteratorAny(mAddressedController.value(), rgbOrder, rgbw, rgbww);
119 }
120
121#if FL_COLOR_PROFILE_RUNTIME
122 const ColorPipelineHooks& hooks = colorPipelineHooks();
123 if (pipeline != nullptr && hooks.makeIterator != nullptr) {
124 // Colour-managed: build the iterator over a streaming source
125 // instead of over the order-variant controller. The source
126 // applies the colour order itself, so it wraps the RGB-ordered
127 // controller -- whichever of the two that is after addressing.
128 PixelController<RGB, 1, 0xFFFFFFFF>& base =
129 mAddressedController ? mAddressedController.value() : pixels;
130 mManagedIterator = hooks.makeIterator(
131 mManagedSourceStorage, mManagedIteratorStorage, base, rgbOrder,
132 *pipeline, rgbw, rgbww, ditherPhase, brightness);
133 }
134#endif
135 }

454bool Channel::reconcileColorProfile(const ChannelOptions& options) FL_NO_EXCEPT {
455 // A strict-mode rejection disables the channel. Remember whether the
456 // current disabled state is ours, so that withdrawing the rejection
457 // re-enables the channel without also overriding a caller's own
458 // setEnabled(false).
459 const bool wasRejectedByUs = mColorProfileFallback && !mProfileBindingAccepted;
460
461 // Recompute from scratch. Reconfiguration must not inherit the verdict of
462 // the configuration it replaces: a valid profile applied over a rejected
463 // one has to clear the fallback, not keep reporting it.
464 mColorProfileFallback = false;
465 mProfileBindingAccepted = true;
466
467 // C5's fallback, raised by the two conditions that mean the same thing to
468 // a caller: management was asked for and is not happening. The second one
469 // -- a profile that binds but builds no pipeline -- is raised further
470 // down, once the build has been attempted.
471 if (options.mColorProfile.mRequested && !options.hasColorProfile()) {
472 raiseColorProfileFallback("no profile bound");
473 }
474 if (options.mColorProfile.mUseGlobalSourceDefault) {
475 mSettings.mColorProfile.mSource = detail::defaultSourceProfile();
476 }
477 // Guards the build below: a binding already rejected by strict mode must
478 // not be built. It is deliberately *not* reused for the enablement
479 // branch -- the build can raise the fallback itself (#4345), and reading
480 // this stale value there left a strict-mode channel reporting Rejected
481 // while still enabled and rendering through the legacy path, which is the
482 // one thing strict mode exists to prevent.
483 const bool rejectedBeforeBuild = mColorProfileFallback && !mProfileBindingAccepted;
484
485 // Derive the pipeline once, here, rather than per frame: it inverts
486 // matrices and bisects a lightness bound. A binding that does not
487 // describe a usable pipeline leaves the channel on the legacy path
488 // rather than failing -- an unbound channel is the ordinary case, not an
489 // error.
490 mPipeline.reset();
491 const ColorPipelineHooks& hooks = colorPipelineHooks();
492 if (!rejectedBeforeBuild && hooks.build != nullptr) {
493 // Through the hook, not by name. Calling `buildPipelineForBinding`
494 // directly from here is what kept the entire pipeline alive in every
495 // build; the pointer is null until `setColorProfile` installs it.
496 // Built on the stack first, so a binding that does not describe a
497 // usable pipeline costs no allocation -- that is the failing half of
498 // an ordinary unbound channel, not an exceptional path.
499 StreamingPipelineQ16 pipeline;
500 if (hooks.build(mSettings.mColorProfile, mSettings, mChipset,
501 &pipeline)) {
502#if FL_COLOR_PIPELINE_SHARED
503 mPipeline = fl::make_shared<StreamingPipelineQ16>(pipeline);
504#else
505 mPipeline = fl::make_unique<StreamingPipelineQ16>(pipeline);
506#endif
507 } else if (options.hasColorProfile()) {
508 // R6: a profile outside the numerical bounds must "fail
509 // explicitly" and "not become native-drive input". A binding that
510 // cannot produce a pipeline used to leave every reporting surface
511 // saying the channel was fine -- Configured, no fallback flag, no
512 // warning -- while it rendered through the legacy path (#4345).
513 //
514 // This reports; it does not reject. Refusing the binding at
515 // setColorProfile() time is R6's fuller reading and needs the
516 // published coefficient bounds it also asks for, which do not
517 // exist yet.
518 raiseColorProfileFallback("bound profile builds no usable pipeline");
519 }
520 }
521
522 // Recomputed after the build, so a rejection raised by a failed build
523 // reaches this branch.
524 const bool rejectedNow = mColorProfileFallback && !mProfileBindingAccepted;
525 if (rejectedNow) {
526 setEnabled(false);
527 } else if (wasRejectedByUs) {
528 setEnabled(true);
529 }
530 return mColorProfileFallback;
531}

575void writeUCS7604(fl::vector_psram<u8>* data, PixelIterator& pixelIterator,
576 ClocklessEncoder encoder, const ChannelOptions& settings,
577 EOrder rgbOrder, bool managed) FL_NO_EXCEPT {
578 // Map encoder enum to UCS7604Mode
579 UCS7604Mode mode;
580 switch (encoder) {
581 case ClocklessEncoder::CLOCKLESS_ENCODER_UCS7604_8BIT:
582 mode = UCS7604Mode::UCS7604_MODE_8BIT_800KHZ;
583 break;
584 case ClocklessEncoder::CLOCKLESS_ENCODER_UCS7604_16BIT:
585 mode = UCS7604Mode::UCS7604_MODE_16BIT_800KHZ;
586 break;
587 case ClocklessEncoder::CLOCKLESS_ENCODER_UCS7604_16BIT_1600:
588 mode = UCS7604Mode::UCS7604_MODE_16BIT_1600KHZ;
589 break;
590 default:
591 // Should never happen â€” caller already checked
592 mode = UCS7604Mode::UCS7604_MODE_16BIT_800KHZ;
593 break;
594 }
595
596 // Get current control from global UCS7604 brightness
597 UCS7604CurrentControl current = ucs7604::brightness();
598
599 // Reorder current control values to match wire order (same logic as UCS7604ControllerT)
600 u8 rgb_currents[3] = {current.r, current.g, current.b};
601 u8 pos0 = (static_cast<int>(rgbOrder) >> 6) & 0x3;
602 u8 pos1 = (static_cast<int>(rgbOrder) >> 3) & 0x3;
603 u8 pos2 = (static_cast<int>(rgbOrder) >> 0) & 0x3;
604 UCS7604CurrentControl wire_current(
605 rgb_currents[pos0], rgb_currents[pos1], rgb_currents[pos2], current.w);
606
607 // Get gamma LUT.
608 //
609 // Only the unbound path uses it. With a profile bound the pixel source
610 // has already produced the device drive and quantized it once, to 16
611 // bits, so widening it again through a curve is the second shaping stage
612 // B1 and section 6 forbid after the device solve -- and the 2.8 here is
613 // `value_or`'s default rather than anything the caller asked for, since
614 // `setColorProfile` clears `mGamma`. #4326.
615 // The iterator's own state, not the options'. `hasColorProfile()` says a
616 // profile is configured; it does not say the managed iterator was built.
617 // A rejected binding leaves the legacy controller in place, and taking
618 // the no-gamma path over its 8-bit pixels would drop a stage that path
619 // still needs.
620 const bool wide_source = managed;
621 float gamma = settings.mGamma.value_or(2.8f);
622 fl::shared_ptr<const Gamma8> gamma8 = Gamma8::getOrCreate(gamma);
623
624 bool is_rgbw = pixelIterator.get_rgbw().active();
625 size_t num_leds = pixelIterator.size();
626
627 // (#2558) UCS7604 is a 4-channel chipset (always RGBW). If the user
628 // configured this channel for 5-channel RGBWW (warm-W + cool-W), the
629 // chipset can't carry the second W byte. Warn loudly â€” silently dropping
630 // the white channels would be very hard to diagnose. The pixel iterator
631 // continues to emit RGB-only bytes (is_rgbw=false) so the strip still
632 // lights up, just without the W diode.
633 if (pixelIterator.get_rgbww().active() && !is_rgbw) {
634 FL_WARN_ONCE("UCS7604 cannot carry 5-channel RGBWW â€” this chipset "
635 "always emits 4-channel RGBW. The warm/cool white "
636 "channels in your Rgbww config will be dropped. "
637 "Set ChannelOptions.mWhiteCfg = Rgbw{...} instead to "
638 "silence this warning.");
639 }
640

830#if defined(FASTLED_DISABLE_DYNAMIC_DRIVER) && FASTLED_DISABLE_DYNAMIC_DRIVER
831 // Body excluded via FASTLED_DISABLE_DYNAMIC_DRIVER (#2926). The
832 // showPixels call site is also gated so this is unreachable; the
833 // empty body lets the linker dead-strip ChannelManager::findDriverByName
834 // and selectDriverForChannel along with this symbol.
835 return {};
836#else
837 // Build busKey only when we actually need it (mBus != AUTO).
838 fl::string busKey;
839 if (mBus != Bus::AUTO) {
840 busKey = fl::string::from_literal(busDriverName(mBus, mBusWhich, isSpi()));
841 }
842
843 fl::shared_ptr<IChannelDriver> driver =
844 ChannelManager::instance().selectDriverForChannel(mChannelData, busKey);
845 mDriver = driver;
846
847 // #2455 / #2459: one-shot diagnostic when a typed-Bus miss happens.
848 // Probe via `findDriverByName` (silent) to distinguish "driver wasn't
849 // instantiated" from "driver exists but canHandle() rejected this
850 // chipset" â€” resolution paths differ. The mBusWarned guard suppresses
851 // duplicate logs on subsequent shows of the same channel.
852#if FASTLED_LOG_RUNTIME_ENABLED
853 if (mBus != Bus::AUTO && !mBusWarned &&
854 (!driver || driver->getName() != busKey)) {
855 auto busDriver = ChannelManager::instance().findDriverByName(busKey);
856 if (!busDriver) {
857 // Typed Bus miss â€” emit the actionable hint with the three
858 // currently-shipping remediations (option 3 added in #2460).
859 FL_ERROR("Channel '" << mName << "': Driver '" << busKey << "' wasn't instantiated. (legacy API; pins Bus + triggers linker keep-alive). Defaulting to AUTO/priority dispatch.");
860 } else {
861 // Registered, but canHandle() said no â€” bus/chipset mismatch.
862 FL_ERROR("Channel '" << mName << "': Driver '" << busKey << "' is registered but cannot handle this channel's chipset (bus/chipset mismatch). Defaulting to AUTO/priority dispatch.");
863 }
864 mBusWarned = true;
865 }
866 if (!driver) {
867 FL_ERROR("Channel '" << mName << "': No compatible driver found - cannot transmit");
868 }
869#endif // FASTLED_LOG_RUNTIME_ENABLED â€” release skips the per-frame
870 // driver->getName() != busKey compare + the silent
871 // findDriverByName probe (used only to differentiate the
872 // FL_ERROR message). The functional return value below is
873 // preserved in both builds. See #2952.
874 return driver;
875#endif // !FASTLED_DISABLE_DYNAMIC_DRIVER

932// Encode `pixels` into mChannelData with the construction-bound encoder and
933// fire onChannelDataEncoded (#4566: out of line from showPixels()).
934void Channel::encodeFrame(PixelController<RGB, 1, 0xFFFFFFFF> &pixels) FL_NO_EXCEPT {
935 // Build pixel iterator with optional addressing transformation
936 // (#2558) Pass both Rgbw and Rgbww from the current rendering settings; the iterator
937 // carries both, and the encoder dispatch below picks the right path based
938 // on which white-channel variant the controller currently holds.
939#if FL_COLOR_PROFILE_RUNTIME
940 // Brightness reaches the pipeline as C4's flux scalar and nowhere else.
941 // `premixed` is the brightness on this path: binding a profile sets
942 // correction and temperature to identity, so nothing else is folded into
943 // it -- and reading it rather than `ColorAdjustment::brightness` keeps
944 // this working when FASTLED_HD_COLOR_MIXING is off, where that field
945 // does not exist.
946 //
947 // The shared pipeline is never written after it is built (#4440): this
948 // managed iterator takes a reference, copies it, and sets flux on the copy, so a
949 // concurrent reader -- the power estimate copies the same pipeline --
950 // never sees a half-written flux, and a reconfiguration replaces
951 // `mPipeline` without touching what this frame encodes through. Small
952 // tiers encode synchronously and keep single ownership, mutated in place.
953#if FL_COLOR_PIPELINE_SHARED
954 const ColorPipelineStorage pipeline_ref = mPipeline;
955 const StreamingPipelineQ16* pipeline = pipeline_ref.get();
956#else
957 StreamingPipelineQ16* pipeline = mPipeline.get();
958#endif
959#else
960 StreamingPipelineQ16* pipeline = nullptr;
961#endif
962 // Dither advances on presentation, not on attempt (#4347, R8): this
963 // channel's phase moves only when its driver accepts a frame (submitFrame()), so
964 // a dropped submission -- no driver, a disabled one, a busy buffer --
965 // does not consume a phase and the cycle's average stays unbiased. Both
966 // the legacy offsets and the colour-managed temporal dither read it.
967 if (pixels.ditherActive()) {
968 pixels.reseed_binary_dithering(mDitherPhase);
969 }
970#ifdef FL_IS_AVR
971 // AVR omits base settings notifications, so base setters are authoritative.
972 const auto& whiteSettings = mLegacySettings;
973#else
974 const auto& whiteSettings = mSettings;
975#endif
976 ReorderingPixelIteratorAny iterator(pixels, mScreenMap.getXYMap(), mRgbOrder,
977 whiteSettings.rgbw(), whiteSettings.rgbww(),
978 mName, pipeline, mDitherPhase,
979 pixels.mColorAdjustment.premixed.r);
980 PixelIterator& pixelIterator = iterator.get();
981
982 // Encode pixels with the writer selected once from the immutable chipset.
983 auto& data = mChannelData->getData();
984 data.clear();
985 const auto* rgbww = whiteSettings.mWhiteCfg.ptr<Rgbww>();
986 const auto* rgbw = whiteSettings.mWhiteCfg.ptr<Rgbw>();
987 if (rgbww != nullptr && rgbww->active()) {
988 mChannelData->setPixelFormat(ChannelPixelFormat::RGBWW);
989 } else if (rgbw != nullptr && rgbw->active()) {
990 mChannelData->setPixelFormat(ChannelPixelFormat::RGBW);
991 } else {
992 mChannelData->setPixelFormat(ChannelPixelFormat::RGB);
993 }
994
995 if (mPixelEncoder != nullptr) {
996 mPixelEncoder(*this, pixelIterator, iterator.isManaged(), data);
997 }
998
999 // Fire event after encoding completes
1000 {
1001 auto& events = ChannelEvents::instance();
1002 events.onChannelDataEncoded(*this, *mChannelData);
1003 }
1004}

1006// Submit the encoded frame to `driver` (#4566: out of line from showPixels()).
1007void Channel::submitFrame(const fl::shared_ptr<IChannelDriver>& driver) FL_NO_EXCEPT {
1008 // #2517: detect the silent-drop scenario before enqueuing â€” if the
1009 // resolved driver is registered with ChannelManager but currently
1010 // disabled (typically by `FastLED.setExclusiveDriver<OtherBus>()`),
1011 // `ChannelManager::onEndFrame()` will skip its `show()` call and the
1012 // frame is silently dropped on the floor. Emit an actionable
1013 // FL_ERROR so users can diagnose the missing output.
1014 //
1015 // One-shot via `mDisabledDriverWarned` to avoid per-frame spam.
1016 // Reset the latch whenever the driver returns to ENABLED so a later
1017 // disable re-emits the diagnostic.
1018 fl::string driverName = driver->getName();
1019 auto status = ChannelManager::instance().driverStatus(driverName);
1020 if (status == ChannelManager::DriverStatus::STATUS_DISABLED) {
1021#if FASTLED_LOG_RUNTIME_ENABLED
1022 if (!mDisabledDriverWarned) {
1023 emitDisabledDriverError(
1024 mName, driverName,
1025 ChannelManager::instance().exclusiveDriverName());
1026 mDisabledDriverWarned = true;
1027 }
1028#endif
1029 // Skip the enqueue â€” the data wouldn't be sent anyway, and dropping
1030 // it here matches the existing behaviour (rather than leaving stale
1031 // bytes in the driver's pending queue across an enable/disable flip).
1032 // The drop happens in both debug and release; only the diagnostic
1033 // and the one-shot latch are gated. In release, the empty
1034 // emitDisabledDriverError + the exclusiveDriverName() call + the
1035 // 3-arg fl::string chain dead-strip (see #2950).
1036 return;
1037 } else if (status == ChannelManager::DriverStatus::STATUS_ENABLED) {
1038#if FASTLED_LOG_RUNTIME_ENABLED
1039 // Reset the one-shot guard so a future disable re-emits the diagnostic.
1040 mDisabledDriverWarned = false;
1041#endif
1042 }
1043 // NOT_REGISTERED: driver is not managed by ChannelManager (e.g. a custom
1044 // test driver, or one that was removed). Fall through to enqueue and let
1045 // the driver decide what to do â€” this is the historic behaviour.
1046
1047 // Enqueue for transmission (will be sent when driver->show() is called)
1048 driver->enqueue(mChannelData);
1049 // Presented, per the driver contract: an accepted enqueue is the frame.
1050 ++mDitherPhase;
1051 auto& events = ChannelEvents::instance();
1052 // Reuse the name fetched for driverStatus() instead of a second getName().
1053 events.onChannelEnqueued(*this, driverName);
1054}
