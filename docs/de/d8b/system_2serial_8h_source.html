283 if (!format) {
284 return 0;
285 }
286
287 // Format into a fixed-size buffer using fl::snprintf

289 int len = fl::snprintf(buffer, sizeof(buffer), format, args...);

291 if (len < 0) {
292 return 0; // Formatting error
293 }
294
295 // Delegate to print() method
296 return print(buffer);
