30 bool decodeNextFrameIfNeeded() FL_NO_EXCEPT {
31 if (mCurrentPos >= mFrameSize || !mHasValidFrame) {
32 if (!mDecoder->hasMoreFrames()) { mHasValidFrame = false; return false; }
33 DecodeResult result = mDecoder->decode();
34 if (result == DecodeResult::Success) {
35 Frame decodedFrame = mDecoder->getCurrentFrame();
36 mCurrentFrame = fl::make_shared<Frame>(decodedFrame);
37 mCurrentPos = 0; mHasValidFrame = true; return true;
38 } else { mHasValidFrame = false; return false; }
39 }
40 return mHasValidFrame;
41 }

44 Mpeg1FileHandle(IDecoderPtr decoder, fl::size pixelsPerFrame, const char* path)
45 FL_NO_EXCEPT : mDecoder(decoder), mCurrentFrame(nullptr), mFrameSize(pixelsPerFrame * 3),
46 mCurrentPos(0), mPath(path), mHasValidFrame(false) { decodeNextFrameIfNeeded(); }
47 bool is_open() const FL_NO_EXCEPT override { return mDecoder != nullptr; }
48 void close() FL_NO_EXCEPT override { if (mDecoder) mDecoder->end(); }
49 fl::size_t read(char* dst, fl::size_t bytesToRead) FL_NO_EXCEPT override {
50 if (!mDecoder || !mHasValidFrame) return 0;
51 fl::size totalRead = 0;
52 while (bytesToRead > 0 && mHasValidFrame) {
53 fl::size rem = mFrameSize - mCurrentPos;
54 if (rem == 0) { if (!decodeNextFrameIfNeeded()) break; rem = mFrameSize - mCurrentPos; }
55 fl::size toRead = fl::min(bytesToRead, rem);
56 if (toRead > 0 && mCurrentFrame && mCurrentFrame->rgb().data()) {
57 fl::memcpy(dst + totalRead, (fl::u8*)mCurrentFrame->rgb().data() + mCurrentPos, toRead);
58 mCurrentPos += toRead; totalRead += toRead; bytesToRead -= toRead;
59 } else break;
60 }
61 return totalRead;
62 }
63 using filebuf::read;

65 fl::size_t tell() FL_NO_EXCEPT override { return 0; }

67 using filebuf::seek;
68 fl::size_t size() const FL_NO_EXCEPT override { return 0; }
69 const char* path() const FL_NO_EXCEPT override { return mPath.c_str(); }

77 if (!mHasValidFrame) return 0;

82Video FileSystem::openMpeg1Video(const char *path, fl::size pixelsPerFrame, float fps,
83 fl::size nFrameHistory) FL_NO_EXCEPT {
84 Video video(pixelsPerFrame, fps, nFrameHistory);
85 fl::ifstream file = openRead(path);
86 if (!file.is_open()) { video.setError(fl::string("Could not open MPEG1 file: ").append(path)); return video; }
87 Mpeg1Config config;
88 config.mode = Mpeg1Config::Streaming;
89 config.targetFps = static_cast<fl::u16>(fps);
90 config.looping = false;
91 config.skipAudio = true;
92 fl::string error_message;
93 IDecoderPtr decoder = Mpeg1::createDecoder(config, &error_message);
94 if (!decoder) { video.setError(fl::string("Failed to create MPEG1 decoder: ").append(error_message)); return video; }
95 if (!decoder->begin(file.rdbuf())) {
96 fl::string decoder_error; decoder->hasError(&decoder_error);
97 video.setError(fl::string("Failed to initialize MPEG1 decoder: ").append(decoder_error)); return video;
98 }
99 fl::shared_ptr<Mpeg1FileHandle> mpeg1Stream = fl::make_shared<Mpeg1FileHandle>(decoder, pixelsPerFrame, path);
100 if (!video.begin(mpeg1Stream)) { video.setError(fl::string("Failed to initialize video with MPEG1 stream")); return video; }
101 return video;
102}

104FramePtr FileSystem::loadJpeg(const char *path, const JpegConfig &config,
105 fl::string *error_message) FL_NO_EXCEPT {
106 fl::ifstream file = openRead(path);
107 if (!file.is_open()) {
108 if (error_message) { *error_message = "Failed to open file: "; error_message->append(path); }
109 FL_WARN("Failed to open JPEG file: " << path); return FramePtr();
110 }
111 fl::size fileSize = file.size();
112 if (fileSize == 0) {
113 if (error_message) { *error_message = "File is empty: "; error_message->append(path); }
114 file.close(); return FramePtr();
115 }
116 fl::vector<u8> buffer;
117 buffer.reserve(fileSize); buffer.resize(fileSize);
118 fl::size bytesRead = 0;
119 while (bytesRead < fileSize && file.available()) {
120 fl::size chunkSize = min<fl::size>(4096, fileSize - bytesRead);
121 fl::size n = file.read(buffer.data() + bytesRead, chunkSize);
122 if (n == 0) break;
123 bytesRead += n;
124 }
125 file.close();
126 if (bytesRead != fileSize) {
127 if (error_message) {
128 *error_message = "Failed to read complete file. Expected ";
129 error_message->append(static_cast<u32>(fileSize));
130 error_message->append(" bytes, got ");
131 error_message->append(static_cast<u32>(bytesRead));
132 }
133 FL_WARN("Failed to read complete JPEG file: " << path); return FramePtr();
134 }
135 fl::span<const u8> jpegData(buffer.data(), buffer.size());
136 FramePtr frame = Jpeg::decode(config, jpegData, error_message);
137 if (!frame && error_message && error_message->empty()) {
138 *error_message = "Failed to decode JPEG from file: "; error_message->append(path);
139 }
140 return frame;
141}
