60void FastLEDAdapter::clear(bool writeToStrip) FL_NO_EXCEPT {
61 CLEDController& controller = FastLED[mControllerIndex];
62 CRGB* leds = controller.leds();
63 if (!leds) {
64 return;
65 }
66
67 if (mHasSegment) {
68 // Clear only the segment
69 for (size_t i = mSegmentStart; i < mSegmentEnd; i++) {
70 leds[i] = CRGB::Black;
71 }
72 } else {
73 // Clear all LEDs
74 size_t numLeds = controller.size();
75 for (size_t i = 0; i < numLeds; i++) {
76 leds[i] = CRGB::Black;
77 }
78 }
79
80 if (writeToStrip) {
81 FastLED.show();
82 }
83}

115void FastLEDAdapter::setSegment(size_t start, size_t end) FL_NO_EXCEPT {
116 CLEDController& controller = FastLED[mControllerIndex];
117 size_t numLeds = controller.size();
118
119 // Validate bounds
120 if (start >= numLeds) {
121 start = numLeds > 0 ? numLeds - 1 : 0;
122 }
123 if (end > numLeds) {
124 end = numLeds;
125 }
126 if (end <= start) {
127 end = start + 1;
128 if (end > numLeds) {
129 end = numLeds;
130 start = end > 0 ? end - 1 : 0;
131 }
132 }
133
134 mSegmentStart = start;
135 mSegmentEnd = end;
136 mHasSegment = true;
137}
