110void NoisePalette::mapNoiseToLEDsUsingPalette(fl::span<CRGB> leds) FL_NO_EXCEPT {
111 static u8 ihue = 0; // okay static in header
112
113 for (u16 i = 0; i < width; i++) {
114 for (u16 j = 0; j < height; j++) {
115 // We use the value at the (i,j) coordinate in the noise
116 // array for our brightness, and the flipped value from (j,i)
117 // for our pixel's index into the color palette.
118
119 u8 index = noise[i * height + j];
120 u8 bri = noise[j * width + i];
121
122 // if this palette is a 'loop', add a slowly-changing base value
123 if (colorLoop) {
124 index += ihue;
125 }
126
127 // brighten up, as the color palette itself often contains the
128 // light/dark dynamic range desired
129 if (bri > 127) {
130 bri = 255;
131 } else {
132 bri = dim8_raw(bri * 2);
133 }
134
135 CRGB color = ColorFromPalette(currentPalette, index, bri);
136 leds[XY(i, j)] = color;
137 }
138 }
139
140 ihue += 1;
141}

143void NoisePalette::fillnoise8() FL_NO_EXCEPT {
144 // If we're running at a low "speed", some 8-bit artifacts become
145 // visible from frame-to-frame. In order to reduce this, we can do some
146 // fast data-smoothing. The amount of data smoothing we're doing depends
147 // on "speed".
148 u8 dataSmoothing = 0;
149 if (speed < 50) {
150 dataSmoothing = 200 - (speed * 4);
151 }
152
153 for (u16 i = 0; i < width; i++) {
154 int ioffset = scale * i;
155 for (u16 j = 0; j < height; j++) {
156 int joffset = scale * j;
157
158 u8 data = inoise8(mX + ioffset, mY + joffset, mZ);
159
160 // The range of the inoise8 function is roughly 16-238.
161 // These two operations expand those values out to roughly
162 // 0..255 You can comment them out if you want the raw noise
163 // data.
164 data = qsub8(data, 16);
165 data = qadd8(data, scale8(data, 39));
166
167 if (dataSmoothing) {
168 u8 olddata = noise[i * height + j];
169 u8 newdata = scale8(olddata, dataSmoothing) +
170 scale8(data, 256 - dataSmoothing);
171 data = newdata;
172 }
173
174 noise[i * height + j] = data;
175 }
176 }
177
178 mZ += speed;
179
180 // apply slow drift to X and Y, just for visual variation.
181 mX += speed / 8;
182 mY -= speed / 16;
183}
