57void HttpConnection::onConnected(u32 currentTimeMs) FL_NO_EXCEPT {
58 if (mState == ConnectionState::CONNECTING || mState == ConnectionState::RECONNECTING) {
59 // Reset attempts if we were in RECONNECTING state before entering CONNECTING
60 // This detects successful reconnection after disconnect
61 bool shouldResetAttempts = mWasReconnecting || (mState == ConnectionState::RECONNECTING);
62
63 transitionTo(ConnectionState::CONNECTED, currentTimeMs);
64
65 if (shouldResetAttempts) {
66 resetReconnectAttempts(); // Successful reconnection, reset counter
67 } else {
68 resetReconnectState(); // Initial connection, just reset timing
69 }
70
71 mWasReconnecting = false;
72 }

162void HttpConnection::transitionTo(ConnectionState newState, u32 currentTimeMs) FL_NO_EXCEPT {
163 mState = newState;
164
165 // Handle state-specific initialization
166 switch (newState) {
167 case ConnectionState::DISCONNECTED:
168 resetReconnectAttempts(); // Full reset on user disconnect
169 break;
170
171 case ConnectionState::CONNECTING:
172 // Starting connection attempt
173 break;
174
175 case ConnectionState::CONNECTED:
176 // Initialize data received timestamp (for timeout detection)
177 mLastDataReceivedMs = currentTimeMs;
178 // Initialize heartbeat timestamp to trigger immediate send on first check
179 // Subtract interval from currentTimeMs to ensure timeSince >= interval
180 if (mLastHeartbeatSentMs == 0) {
181 // Use wrapping arithmetic: currentTimeMs - interval
182 // Even if this underflows (currentTimeMs < interval), the unsigned
183 // subtraction in shouldSendHeartbeat will wrap correctly
184 mLastHeartbeatSentMs = currentTimeMs - mConfig.heartbeatIntervalMs;
185 }
186 break;
187
188 case ConnectionState::RECONNECTING:
189 // Calculate backoff delay BEFORE incrementing attempts
190 mReconnectDelayMs = calculateBackoffDelay();
191 mNextReconnectTimeMs = currentTimeMs + mReconnectDelayMs;
192 // Increment attempts for next reconnection
193 mReconnectAttempts++;
194 break;
195
196 case ConnectionState::CLOSED:
197 resetReconnectAttempts(); // Full reset on close
198 break;
199 }
200}

214u32 HttpConnection::calculateBackoffDelay() const FL_NO_EXCEPT {
215 // Exponential backoff: delay = initial * (multiplier ^ attempts)
216 u32 delay = mConfig.reconnectInitialDelayMs;
217 for (u32 i = 0; i < mReconnectAttempts; i++) {
218 delay *= mConfig.reconnectBackoffMultiplier;
219 if (delay > mConfig.reconnectMaxDelayMs) {
220 delay = mConfig.reconnectMaxDelayMs;
221 break;
222 }
223 }
224
225 return delay;
226}
