464 FASTLED_FORCE_INLINE pixelset_iterator_base& operator++() FL_NO_EXCEPT { leds += dir; return *this; }
465 FASTLED_FORCE_INLINE pixelset_iterator_base operator++(int) FL_NO_EXCEPT { pixelset_iterator_base tmp(*this); leds += dir; return tmp; }
466
467 // `const &`, not `&`: taking a mutable lvalue reference meant these
468 // could not bind a temporary, so the idiomatic
469 // `for (it = s.begin(); it != s.end(); ++it)` failed to compile
470 // against the rvalue returned by end(). Every loop in this file works
471 // around that by hoisting `end()` into a named variable first. #822.
472 FASTLED_FORCE_INLINE bool operator==(const pixelset_iterator_base & other) const FL_NO_EXCEPT { return leds == other.leds; /* && set==other.set; */ }
473 FASTLED_FORCE_INLINE bool operator!=(const pixelset_iterator_base & other) const FL_NO_EXCEPT { return leds != other.leds; /* || set != other.set; */ }
474
475 FASTLED_FORCE_INLINE PIXEL_TYPE& operator*() const FL_NO_EXCEPT { return *leds; }
476 };
