26// Polar coordinate pre-computation
27// Builds polar_theta[x][y] and distance[x][y] lookup tables
28inline void render_polar_lookup_table(float cx, float cy,
29 fl::vector<fl::vector<float>> &polar_theta,
30 fl::vector<fl::vector<float>> &distance,
31 int num_x, int num_y) FL_NO_EXCEPT {
32 polar_theta.resize(num_x, fl::vector<float>(num_y, 0.0f));
33 distance.resize(num_x, fl::vector<float>(num_y, 0.0f));
34
35 for (int xx = 0; xx < num_x; xx++) {
36 for (int yy = 0; yy < num_y; yy++) {
37 float dx = xx - cx;
38 float dy = yy - cy;
39 distance[xx][yy] = fl::hypotf(dx, dy);
40 polar_theta[xx][yy] = fl::atan2f(dy, dx);
41 }
42 }
43}

45// Calculate oscillators from timing ratios
46// Computes linear, radial, directional, and noise_angle modulators
47inline void calculate_oscillators(oscillators &timings, modulators &move,
48 fl::u32 current_time, float speed_factor) FL_NO_EXCEPT {
49 double runtime = current_time * timings.master_speed * speed_factor;
50
51 for (int i = 0; i < num_oscillators; i++) {
52 move.linear[i] = (runtime + timings.offset[i]) * timings.ratio[i];
53 move.radial[i] = fl::fmodf(move.linear[i], 2.0f * static_cast<float>(PI));
54 move.directional[i] = FL_SIN_F(move.radial[i]);
55 move.noise_angle[i] = PI * (1 + pnoise(move.linear[i], 0, 0));
56 }
57}

91FASTLED_FORCE_INLINE float map_float(float x, float in_min, float in_max, float out_min,
92 float out_max) FL_NO_EXCEPT {
93 float result =
94 (x - in_min) * (out_max - out_min) / (in_max - in_min) + out_min;
95 if (result < out_min)
96 result = out_min;
97 if (result > out_max)
98 result = out_max;
99 return result;
100}

102// Main noise field renderer with clamping
103// Converts polar coordinates to cartesian, applies transformations, and renders Perlin noise
104FASTLED_FORCE_INLINE float render_value(render_parameters &animation) FL_NO_EXCEPT {
105 // Convert polar coordinates back to cartesian ones
106 float newx = (animation.offset_x + animation.center_x -
107 (FL_COS_F(animation.angle) * animation.dist)) *
108 animation.scale_x;
109 float newy = (animation.offset_y + animation.center_y -
110 (FL_SIN_F(animation.angle) * animation.dist)) *
111 animation.scale_y;
112 float newz = (animation.offset_z + animation.z) * animation.scale_z;
113
114 // Render noise value at this new cartesian point
115 float raw_noise_field_value = pnoise(newx, newy, newz);
116
117 // Enhance histogram (improve contrast) by setting black and white point
118 if (raw_noise_field_value < animation.low_limit)
119 raw_noise_field_value = animation.low_limit;
120 if (raw_noise_field_value > animation.high_limit)
121 raw_noise_field_value = animation.high_limit;
122
123 float scaled_noise_value =
124 map_float(raw_noise_field_value, animation.low_limit,
125 animation.high_limit, 0, 255);
126
127 return scaled_noise_value;
128}
