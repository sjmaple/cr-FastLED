72enum class PwmBackend : u8 {
73 None, // No setPwmFrequency called — use platform default
74 Native, // Platform hardware PWM handles frequency
75 IsrSoftware // ISR-based software PWM
76};

78constexpr u8 MAX_PWM_CHANNELS = 8;
79constexpr u32 ISR_FREQUENCY_HZ = 128000; // 128 kHz for 8-bit @ 500 Hz
80constexpr u32 MAX_ISR_PWM_FREQUENCY = 500;

82struct PwmPinState {
83 int pin; // GPIO pin (-1 = unused)
84 u32 frequency_hz; // Configured PWM frequency
85 PwmBackend backend; // Which backend handles this pin
86 u8 duty_cycle; // 0-255 (ISR backend duty)
87 // ISR-only fields:
88 u16 period_ticks; // ISR ticks per PWM period
89 u16 high_ticks; // Ticks to stay HIGH
90 u16 tick_counter; // Current tick (0 to period_ticks-1)
91 bool pin_state; // Current GPIO state (ISR only)
92
93 PwmPinState() FL_NO_EXCEPT : pin(-1), frequency_hz(0), backend(PwmBackend::None),
94 duty_cycle(0), period_ticks(0), high_ticks(0),
95 tick_counter(0), pin_state(false) {}
96};

98// Singleton state container for PWM management
99struct PwmStateData {
100 PwmPinState channels[MAX_PWM_CHANNELS];
101 fl::isr::handle isr_handle;
102 bool isr_active;
103
104 PwmStateData() FL_NO_EXCEPT : isr_active(false) {}
105};

107// Set the first time state() runs, which happens only on the paths that
108// allocate a channel (setPwmFrequency) or service one (the ISR). The
109// lookup paths taken by every sketch -- pinMode(), analogWrite(),
110// setPwm16() -- read this pointer instead of calling state(), so a sketch
111// that never calls setPwmFrequency() does not link the singleton storage.
112// A null pointer means no channel has ever been allocated. The target is
113// static storage, so the raw pointer can never dangle.
114// FL_LINT_ALLOW_GLOBAL(constant-initialized null pointer; see above)
115PwmStateData* g_state = nullptr;
116
117// Access singleton state (creates it on first use)
118inline PwmStateData& state() FL_NO_EXCEPT {
119 PwmStateData& st = fl::Singleton<PwmStateData>::instance();
120 g_state = &st;
121 return st;
122}

124// ISR handler — services only ISR-backend entries
125void FL_IRAM pwm_isr_handler(void* user_data) FL_NO_EXCEPT {
126 (void)user_data;
127 // The ISR is attached only after setPwmFrequency() has created the
128 // state, so read the singleton directly instead of republishing g_state.
129 PwmStateData& st = fl::Singleton<PwmStateData>::instance();
130
131 for (u8 i = 0; i < MAX_PWM_CHANNELS; i++) {
132 PwmPinState& ch = st.channels[i];
133 if (ch.pin < 0 || ch.backend != PwmBackend::IsrSoftware) continue;
134
135 ch.tick_counter++;
136
137 // Turn LOW at duty cycle boundary
138 if (ch.tick_counter == ch.high_ticks && ch.pin_state) {
139 fl::digitalWrite(ch.pin, fl::PinValue::Low);
140 ch.pin_state = false;
141 }
142 // Start new period: reset counter, turn HIGH if duty > 0
143 else if (ch.tick_counter >= ch.period_ticks) {
144 ch.tick_counter = 0;
145 if (ch.high_ticks > 0) {
146 fl::digitalWrite(ch.pin, fl::PinValue::High);
147 ch.pin_state = true;
148 }
149 }
150 }
151}

153// Find channel by pin number
154PwmPinState* findByPin(int pin) FL_NO_EXCEPT {
155 if (!g_state) return nullptr; // No channel ever allocated.
156 PwmStateData& st = *g_state;
157 for (u8 i = 0; i < MAX_PWM_CHANNELS; i++) {
158 if (st.channels[i].pin == pin) {
159 return &st.channels[i];
160 }
161 }
162 return nullptr;
163}
164
165// Allocate a free channel
166PwmPinState* allocate() FL_NO_EXCEPT {
167 PwmStateData& st = state();
168 for (u8 i = 0; i < MAX_PWM_CHANNELS; i++) {
169 if (st.channels[i].pin < 0) {
170 return &st.channels[i];
171 }
172 }
173 return nullptr;
174}

176// Count active ISR-backend channels
177u8 countIsrChannels() FL_NO_EXCEPT {
178 if (!g_state) return 0;
179 PwmStateData& st = *g_state;
180 u8 count = 0;
181 for (u8 i = 0; i < MAX_PWM_CHANNELS; i++) {
182 if (st.channels[i].pin >= 0 && st.channels[i].backend == PwmBackend::IsrSoftware) {
183 count++;
184 }
185 }
186 return count;
187}
188
189// Ensure ISR timer is running (lazy init)
190int ensureIsrActive() FL_NO_EXCEPT {
191 PwmStateData& st = state();
192 if (st.isr_active) return 0;
193
194 fl::isr::config cfg;
195 cfg.handler = pwm_isr_handler;
196 cfg.frequency_hz = ISR_FREQUENCY_HZ;
197 cfg.priority = fl::isr::ISR_PRIORITY_MEDIUM;
198 cfg.flags = fl::isr::ISR_FLAG_IRAM_SAFE;
199
200 int result = fl::isr::attach_timer_handler(cfg, &st.isr_handle);
201 if (result != 0) {
202 FL_WARN("PWM: ISR attach failed: " << (fl::isr::get_error_string(result)));
203 return result;
204 }
205 st.isr_active = true;
206 return 0;
207}
208
209// Shutdown ISR if no ISR-backend channels remain
210void maybeShutdownIsr() FL_NO_EXCEPT {
211 if (!g_state) return; // ISR can only be active once state exists.
212 PwmStateData& st = *g_state;
213 if (!st.isr_active) return;
214 if (countIsrChannels() > 0) return;
215
216 fl::isr::detach_handler(st.isr_handle);
217 st.isr_active = false;
218}

220// Release a channel and cleanup
221void releaseChannel(PwmPinState* ch) FL_NO_EXCEPT {
222 if (!ch || ch->pin < 0) return;
223
224 fl::digitalWrite(ch->pin, fl::PinValue::Low);
225
226 {
227 fl::isr::critical_section cs;
228 ch->pin = -1;
229 ch->backend = PwmBackend::None;
230 ch->frequency_hz = 0;
231 ch->duty_cycle = 0;
232 }
233
234 maybeShutdownIsr();
235}

243void analogWrite(int pin, u16 val) FL_NO_EXCEPT {
244 pwm_state::PwmPinState* ch = pwm_state::findByPin(pin);
245
246 if (ch && ch->backend == pwm_state::PwmBackend::IsrSoftware) {
247 // Route to ISR duty cycle update (scale 8-bit val to duty)
248 u8 duty = (val > 255) ? 255 : static_cast<u8>(val);
249 fl::isr::critical_section cs;
250 ch->duty_cycle = duty;
251 ch->high_ticks = (static_cast<u32>(ch->period_ticks) * duty) / 256;
252 return;
253 }
254
255 if (ch && ch->backend == pwm_state::PwmBackend::Native) {
256 // Native backend — forward to platform (frequency already configured)
257 platforms::analogWrite(pin, val);
258 return;
259 }
260
261 // No setPwmFrequency called — forward to platform as before
262 platforms::analogWrite(pin, val);
263}

265void setPwm16(int pin, u16 val) FL_NO_EXCEPT {
266 pwm_state::PwmPinState* ch = pwm_state::findByPin(pin);
267
268 if (ch && ch->backend == pwm_state::PwmBackend::IsrSoftware) {
269 // Route to ISR duty cycle update (scale 16-bit to 8-bit)
270 u8 duty = static_cast<u8>(val >> 8);
271 fl::isr::critical_section cs;
272 ch->duty_cycle = duty;
273 ch->high_ticks = (static_cast<u32>(ch->period_ticks) * duty) / 256;
274 return;
275 }
276
277 if (ch && ch->backend == pwm_state::PwmBackend::Native) {
278 platforms::setPwm16(pin, val);
279 return;
280 }
281
282 // No setPwmFrequency called — forward to platform as before
283 platforms::setPwm16(pin, val);
284}

290int setPwmFrequency(int pin, u32 frequency_hz) FL_NO_EXCEPT {
291 // Check if pin already has a channel
292 pwm_state::PwmPinState* ch = pwm_state::findByPin(pin);
293
294 if (ch) {
295 // Pin already configured — release and reconfigure
296 pwm_state::releaseChannel(ch);
297 ch = nullptr;
298 }
299
300 // Validate frequency
301 if (frequency_hz == 0) {
302 FL_WARN("setPwmFrequency: Frequency must be > 0");
303 return -1;
304 }
305
306 // Query platform: can it handle this natively?
307 bool needs_isr = platforms::needsPwmIsrFallback(pin, frequency_hz);
308
309 if (!needs_isr) {
310 // Native path
311 int result = platforms::setPwmFrequencyNative(pin, frequency_hz);
312 if (result != 0) {
313 FL_WARN("setPwmFrequency: Native backend failed: " << result);
314 return result;
315 }
316
317 // Allocate tracking slot
318 ch = pwm_state::allocate();
319 if (!ch) {
320 FL_WARN("setPwmFrequency: All " << (static_cast<int>(pwm_state::MAX_PWM_CHANNELS)) << " channels in use");
321 return -2;
322 }
323
324 ch->pin = pin;
325 ch->frequency_hz = frequency_hz;
326 ch->backend = pwm_state::PwmBackend::Native;
327 ch->duty_cycle = 0;
328 return 0;
329 }
330
331 // ISR software fallback path
332 if (frequency_hz > pwm_state::MAX_ISR_PWM_FREQUENCY) {
333 FL_WARN("setPwmFrequency: ISR fallback max " << (pwm_state::MAX_ISR_PWM_FREQUENCY) << " Hz, requested " << frequency_hz);
334 return -1;
335 }
336
337 // Allocate channel
338 ch = pwm_state::allocate();
339 if (!ch) {
340 FL_WARN("setPwmFrequency: All " << (static_cast<int>(pwm_state::MAX_PWM_CHANNELS)) << " channels in use");
341 return -2;
342 }
343
344 // Ensure ISR is running
345 int isr_result = pwm_state::ensureIsrActive();
346 if (isr_result != 0) {
347 return -3;
348 }
349
350 // Configure GPIO
351 fl::pinMode(pin, fl::PinMode::Output);
352 fl::digitalWrite(pin, fl::PinValue::Low);
353
354 // Initialize channel (atomic)
355 {
356 fl::isr::critical_section cs;
357 ch->pin = pin;
358 ch->frequency_hz = frequency_hz;
359 ch->backend = pwm_state::PwmBackend::IsrSoftware;
360 ch->period_ticks = pwm_state::ISR_FREQUENCY_HZ / frequency_hz;
361 ch->duty_cycle = 0;
362 ch->high_ticks = 0;
363 ch->tick_counter = 0;
364 ch->pin_state = false;
365 }
366
367 return 0;
368}
